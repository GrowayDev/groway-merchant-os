   323→export function POSLayout({ vertical, subVertical, vasTiles, onBack }: POSLayoutProps) {
   324→  // Offline mode hook - must be used within OfflineModeProvider
   325→  const offlineMode = useOfflineMode()
   326→  
   327→  // Responsive state
   328→  const [screenWidth, setScreenWidth] = useState(1024)
   329→  const [mobileAppMode, setMobileAppMode] = useState(false)
   330→  
   331→  // Core state
   332→  const [currentView, setCurrentView] = useState<ViewType>('dashboard')

   342→

   345→  const [cartDiscount, setCartDiscount] = useState<Discount | null>(null)
   346→  const [heldTransactions, setHeldTransactions] = useState<HeldTransaction[]>([])
   347→  const [completedTransaction, setCompletedTransaction] = useState<{
   348→    id: string
   349→    number: string
   350→    items: CartItem[]
   351→    subtotal: number
   352→    discount?: number
   353→    vat: number
   354→    total: number
   355→    paymentMethod?: string
   356→    customerName?: string
   357→    createdAt: Date
   358→  } | null>(null)
   359→  
   360→  // Dialog states
   361→  const [priceOverrideOpen, setPriceOverrideOpen] = useState(false)
   362→  const [itemDiscountOpen, setItemDiscountOpen] = useState(false)
   363→  const [cartDiscountOpen, setCartDiscountOpen] = useState(false)
   364→  const [splitPaymentOpen, setSplitPaymentOpen] = useState(false)
   365→  const [customerLookupOpen, setCustomerLookupOpen] = useState(false)
   366→  const [holdTransactionOpen, setHoldTransactionOpen] = useState(false)
   367→  const [receiptPreviewOpen, setReceiptPreviewOpen] = useState(false)
   368→  const [voidRefundOpen, setVoidRefundOpen] = useState(false)
   369→  const [voidRefundMode, setVoidRefundMode] = useState<'void' | 'refund'>('void')
   370→  const [selectedItemForEdit, setSelectedItemForEdit] = useState<CartItem | null>(null)
   371→  const [isSplitPayment, setIsSplitPayment] = useState(false)
   372→  const [splitPayments, setSplitPayments] = useState<SplitPayment[]>([])
   373→  
   374→  // Mobile-specific state
   375→  const [mobileCartOpen, setMobileCartOpen] = useState(false)
   376→  const [mobileSearchOpen, setMobileSearchOpen] = useState(false)
   377→  const [mobileMoreOpen, setMobileMoreOpen] = useState(false)
   378→  
   379→  // Responsive breakpoints
   380→  const isMobile = screenWidth < 640 || mobileAppMode
   381→  const isTablet = screenWidth >= 640 && screenWidth < 1024
   382→  const isDesktop = screenWidth >= 1024 && !mobileAppMode
   383→  
   384→  // Track screen size
   385→  useEffect(() => {
   386→    const handleResize = () => setScreenWidth(window.innerWidth)
   387→    handleResize()
   388→    window.addEventListener('resize', handleResize)
   389→    return () => window.removeEventListener('resize', handleResize)
   390→  }, [])
   391→
   392→  // Get SKUs for this vertical/sub-vertical
   393→  const skus = getAllSKUs(vertical.id, subVertical.id)
   394→  
   395→  // Get unique categories
   396→  const categories = ['all', ...new Set(skus.map(sku => sku.category))]
   397→  
   398→  // Filter SKUs by search and category
   399→  const filteredSKUs = skus.filter(sku => {
   400→    const matchesSearch = sku.name.toLowerCase().includes(searchQuery.toLowerCase())
   401→    const matchesCategory = activeCategory === 'all' || sku.category === activeCategory
   402→    return matchesSearch && matchesCategory
   403→  })
   404→
   405→  // Cart calculations with discount and price override support
   406→  const itemsSubtotal = cart.reduce((sum, item) => {
   407→    const effectivePrice = item.priceOverride ?? item.price
   408→    const itemTotal = effectivePrice * item.quantity
   409→    const itemDiscount = item.discount || 0
   410→    return sum + (itemTotal - itemDiscount)
   411→  }, 0)
   412→  
   413→  // Apply cart-level discount
   414→  const cartDiscountAmount = cartDiscount
   415→    ? cartDiscount.type === 'percentage'
   416→      ? itemsSubtotal * (cartDiscount.value / 100)
   417→      : cartDiscount.value
   418→    : 0
   419→  
   420→  const subtotal = itemsSubtotal - cartDiscountAmount
   421→  const vat = subtotal * 0.15 // 15% VAT
   422→  const total = subtotal + vat
   423→
   424→  const addToCart = (sku: typeof skus[0]) => {
   425→    setCart(prev => {
   426→      const existing = prev.find(item => item.name === sku.name)

   428→        return prev.map(item => 
   429→          item.name === sku.name 
   430→            ? { ...item, quantity: item.quantity + 1 }
   431→            : item
   432→        )
   433→      }
   434→      return [...prev, { id: Date.now().toString(), name: sku.name, price: sku.price, quantity: 1, category: sku.category }]

   457→    }).filter(item => item.quantity > 0))

   476→    setCart(prev => prev.map(item => {
   477→      if (item.id === itemId) {
   478→        return {
   479→          ...item,
   480→          priceOverride: newPrice,
   481→          originalPrice: item.originalPrice || item.price,
   482→          notes: reason
   483→        }
   484→      }
   485→      return item
   486→    }))
   487→  }
   488→  
   489→  // Discount handlers
   490→  const handleItemDiscount = (discount: Discount, itemId?: string) => {
   491→    if (!itemId) return
   492→    setCart(prev => prev.map(item => {
   493→      if (item.id === itemId) {
   494→        const effectivePrice = item.priceOverride ?? item.price
   495→        const discountAmount = discount.type === 'percentage'
   496→          ? effectivePrice * item.quantity * (discount.value / 100)
   497→          : discount.value
   498→        return {
   499→          ...item,
   500→          discount: discountAmount,
   501→          discountType: discount.type
   502→        }
   503→      }
   504→      return item
   505→    }))
   506→  }
   507→  
   508→  const handleCartDiscount = (discount: Discount) => {
   509→    setCartDiscount(discount)
   510→  }
   511→  
   512→  // Hold transaction handlers
   513→  const handleHoldTransaction = (notes?: string) => {
   514→    const heldTx: HeldTransaction = {
   515→      id: `held-${Date.now()}`,
   516→      items: [...cart],
   517→      subtotal: itemsSubtotal,
   518→      discount: cartDiscountAmount,
   519→      discountType: cartDiscount?.type || 'fixed',
   520→      notes,
   521→      customerName: selectedCustomer?.name || bookingData?.customerName,
   522→      customerId: selectedCustomer?.id || bookingData?.customerId,
   523→      heldAt: new Date()
   524→    }
   525→    setHeldTransactions(prev => [...prev, heldTx])
   526→    clearCart()
   527→  }
   528→  
   529→  const handleResumeTransaction = (transaction: HeldTransaction) => {
   530→    setCart(transaction.items)
   531→    if (transaction.customerName) {
   532→      setSelectedCustomer({
   533→        id: transaction.customerId || '',
   534→        name: transaction.customerName
   535→      })
   536→    }
   537→    if (transaction.discount > 0) {
   538→      setCartDiscount({
   539→        type: transaction.discountType,
   540→        value: transaction.discount
   541→      })
   542→    }
   543→    setHeldTransactions(prev => prev.filter(t => t.id !== transaction.id))
   544→  }
   545→  
   546→  const handleDeleteHeldTransaction = (id: string) => {
   547→    setHeldTransactions(prev => prev.filter(t => t.id !== id))
   548→  }
   549→  
   550→  // Split payment handler
   551→  const handleSplitPaymentComplete = (payments: SplitPayment[]) => {
   552→    setSplitPayments(payments)
   553→    setIsSplitPayment(true)
   554→    // Process the sale with split payments
   555→    processSale(payments)
   556→  }
   557→  
   558→  // Process sale
   559→  const processSale = async (payments?: SplitPayment[]) => {
   560→    const sessionStr = localStorage.getItem('groway_session')
   561→    const session = sessionStr ? JSON.parse(sessionStr) : null
   562→    const merchantId = session?.merchantId || 'merchant-001'
   563→    const branchId = session?.branchId || 'branch-001'
   564→    
   565→    const transactionNumber = `TXN-${Date.now()}-${Math.random().toString(36).slice(2, 6).toUpperCase()}`
   566→    
   567→    setCheckoutOpen(false)
   568→    setSaleComplete(true)
   569→    setOfflineTransactionSaved(false)
   570→
   571→    // Check if offline
   572→    if (!offlineMode.isOnline) {
   573→      try {
   574→        await offlineMode.addTransaction({
   575→          type: 'sale',
   576→          amount: total,
   577→          items: cart.map(item => ({
