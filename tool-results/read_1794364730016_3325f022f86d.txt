   250→  const locations = stockLocations.slice(0, 6)
   251→  
   252→  return skus.map((sku, index) => {
   253→    const stock = Math.floor(Math.random() * 100) + (index % 5 === 0 ? 0 : 5)
   254→    const cost = sku.price * (0.4 + Math.random() * 0.3)
   255→    const minStock = Math.floor(Math.random() * 15) + 5
   256→    const reorderPoint = minStock * 2
   257→    const velocity = Math.floor(Math.random() * 30) + 5
   258→    const isBatchTracked = index % 3 === 0
   259→    const isExpiryTracked = index % 4 === 0
   260→    
   261→    // Generate batches for batch-tracked items
   262→    const batches: Batch[] = isBatchTracked ? [

   268→        receivedDate: new Date(Date.now() - 30 * 24 * 60 * 60 * 1000).toISOString().split('T')[0],
   269→        cost: cost
   270→      },
   271→      {
   272→        batchNo: `B${Date.now().toString().slice(-6)}-${index + 1}`,
   273→        qty: Math.floor(stock * 0.4),
   274→        expiry: new Date(Date.now() + 15 * 24 * 60 * 60 * 1000).toISOString().split('T')[0],
   275→        status: 'expiring_soon',
   276→        receivedDate: new Date(Date.now() - 60 * 24 * 60 * 60 * 1000).toISOString().split('T')[0],

   279→    ] : undefined
   280→    
   281→    return {
   282→      id: `INV-${String(index + 1).padStart(4, '0')}`,
   283→      name: sku.name,
   284→      category: sku.category,
   285→      sku: `${sku.category.substring(0, 3).toUpperCase()}-${String(index + 1).padStart(3, '0')}`,
   286→      barcode: `600${String(Math.floor(Math.random() * 10000000000)).padStart(11, '0')}`,
   287→      price: sku.price,
   288→      cost: Math.round(cost * 100) / 100,
   289→      stock,
   290→      minStock,
   291→      reorderPoint,
   292→      reorderQty: reorderPoint * 2,
   293→      unit: 'unit',
   294→      location: locations[index % locations.length],
   295→      batchTracked: isBatchTracked,

   308→// Generate movements based on inventory
   309→const generateMovementsFromInventory = (inventory: InventoryItem[]): StockMovement[] => {
   310→  const types: StockMovement['type'][] = ['sale', 'return', 'grn', 'adjustment', 'transfer_in', 'transfer_out', 'stocktake']
   311→  const productNames = inventory.slice(0, 5).map(i => i.name)
   312→  const users = ['Thandi M.', 'Sipho K.', 'System', 'Johan P.', 'Maria S.', 'Peter N.']
   313→  const reasons = ['Stock correction', 'Damaged goods', 'Theft reported', 'Found during stocktake', 'Supplier return', 'Quality issue']
   314→  
   315→  return Array.from({ length: 30 }, (_, i) => {
   316→    const type = types[Math.floor(Math.random() * types.length)]
   317→    const isOut = ['sale', 'transfer_out'].includes(type)
   318→    const randomProduct = productNames[Math.floor(Math.random() * productNames.length)] || 'Product'
   319→    const item = inventory.find(p => p.name === randomProduct)
   320→    
   321→    return {
   322→      id: `MOV-${String(i + 1).padStart(6, '0')}`,
   323→      type,
   324→      direction: isOut ? 'out' : 'in',
   325→      productName: randomProduct,
   326→      sku: item?.sku || `SKU-${String(Math.floor(Math.random() * 1000)).padStart(3, '0')}`,
   327→      quantity: 1 + Math.floor(Math.random() * 50),
   328→      reason: type === 'adjustment' ? reasons[Math.floor(Math.random() * reasons.length)] : undefined,
   329→      user: users[Math.floor(Math.random() * users.length)],
   330→      timestamp: new Date(Date.now() - i * 3600000).toISOString(),
   331→      value: Math.floor(Math.random() * 5000) / 100,
   332→      fromLocation: type === 'transfer_out' ? stockLocations[Math.floor(Math.random() * 5)] : undefined,
   333→      toLocation: type === 'transfer_in' ? stockLocations[Math.floor(Math.random() * 5)] : undefined,
   334→      batchNo: item?.batchTracked ? `B${Date.now().toString().slice(-6)}` : undefined,
   335→    }
   336→  })
   337→}

   339→// Generate GroSupply suggestions based on inventory
   340→const generateGroSupplySuggestionsFromInventory = (inventory: InventoryItem[]): GroSupplySuggestion[] => {
   341→  const lowStockItems = inventory.filter(i => i.stock <= i.reorderPoint).slice(0, 4)
   342→  
   343→  return lowStockItems.map((item, index) => ({
   344→    id: `GS-${String(index + 1).padStart(3, '0')}`,
   345→    productName: item.name,
   346→    sku: item.sku,
   347→    currentStock: item.stock,
   348→    suggestedOrder: item.reorderQty,
   349→    supplier: item.supplier || 'Unknown Supplier',
   350→    leadTime: item.supplierLeadTime || 3,
   351→    estimatedCost: Math.round(item.reorderQty * (item.cost || item.price * 0.5) * 100) / 100,
   352→    reason: item.stock === 0 ? 'Out of stock' : 'Below reorder point',
   353→    urgency: item.stock === 0 ? 'critical' as const : 'high' as const,
   354→    savings: Math.round(Math.random() * 150),
   355→  }))
   356→}

   358→// Generate reorder alerts
   359→const generateReorderAlerts = (inventory: InventoryItem[]): ReorderAlert[] => {
   360→  const alerts: ReorderAlert[] = []
   361→  
   362→  inventory.forEach(item => {
   363→    if (item.stock === 0) {
   364→      alerts.push({
   365→        id: `ALERT-${item.id}-OOS`,
   366→        item,
   367→        type: 'out_of_stock',
   368→        message: `${item.name} is out of stock and needs immediate reorder`,
   369→        createdAt: new Date().toISOString(),
   370→        acknowledged: false
   371→      })
   372→    } else if (item.stock <= item.reorderPoint) {
   373→      alerts.push({
   374→        id: `ALERT-${item.id}-LOW`,
   375→        item,
   376→        type: 'low_stock',
   377→        message: `${item.name} is below reorder point (${item.stock}/${item.reorderPoint})`,
   378→        createdAt: new Date().toISOString(),
   379→        acknowledged: false
   380→      })
   381→    }
   382→    
   383→    // Check for expiring batches
   384→    if (item.batches) {
   385→      item.batches.forEach(batch => {
   386→        if (batch.status === 'expiring_soon') {
   387→          alerts.push({
   388→            id: `ALERT-${item.id}-${batch.batchNo}`,
   389→            item,
   390→            type: 'expiring_soon',
   391→            message: `${item.name} batch ${batch.batchNo} expires on ${batch.expiry}`,
   392→            createdAt: new Date().toISOString(),
   393→            acknowledged: false
   394→          })
   395→        }
   396→      })
   397→    }
   398→  })
   399→  
   400→  return alerts
   401→}
