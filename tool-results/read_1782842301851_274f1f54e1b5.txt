    75→function useAuth() {

   140→  const [branchProvince, setBranchProvince] = useState('')

   142→

   145→    mon: { open: '08:00', close: '17:00', closed: false },

   331→            <div className="flex-1">
   332→              <Progress value={progress} className="h-2" />
   333→            </div>

   343→              >
   344→                <div className={`w-8 h-8 rounded-full flex items-center justify-center ${
   345→                  index < currentStep 
   346→                    ? 'bg-[#00897B] text-white' 
   347→                    : index === currentStep 
   348→                      ? 'bg-[#00897B] text-white' 
   349→                      : 'bg-gray-200 text-[#757575]'

   369→                <CardTitle>{onboardingSteps[currentStep].title}</CardTitle>

   371→              </div>
   372→            </div>
   373→          </CardHeader>
   374→          
   375→          <CardContent className="space-y-6">

   378→              <div className="text-center py-8">
   379→                <div className="w-20 h-20 rounded-full bg-gradient-to-br from-[#00897B] to-[#4CAF50] mx-auto flex items-center justify-center mb-4">
   380→                  <Sparkles className="h-10 w-10 text-white" />

   912→  const [merchant, setMerchant] = useState<AuthMerchant | null>(null)
   913→  const [branchId, setBranchId] = useState<string | null>(null)
   914→  const [role, setRole] = useState<string>('staff')
   915→
   916→  const checkAuth = async () => {
   917→    try {
   918→      const response = await fetch('/api/auth/me', {
   919→        credentials: 'include',
   920→      })
   921→
   922→      if (response.ok) {
   923→        const data = await response.json()
   924→        if (data.success) {
   925→          setUser(data.user)
   926→          setMerchant(data.merchant)
   927→          setBranchId(data.branchId)
   928→          setRole(data.role)
   929→          return { authenticated: true }
   930→        }
   931→      }
   932→      return { authenticated: false }
   933→    } catch (error) {
   934→      console.error('Auth check error:', error)
   935→      return { authenticated: false }
   936→    }
   937→  }
   938→
   939→  const refresh = async () => {
   940→    await checkAuth()
   941→  }
   942→
   943→  const logout = async () => {
   944→    try {
   945→      await fetch('/api/auth/logout', {
   946→        method: 'POST',
   947→        credentials: 'include',
   948→      })
   949→    } catch (error) {
   950→      console.error('Logout error:', error)
   951→    }
   952→    setUser(null)
   953→    setMerchant(null)
   954→    router.push('/merchant-os/login')
   955→  }
   956→
   957→  useEffect(() => {
   958→    const init = async () => {
   959→      const result = await checkAuth()
   960→      
   961→      if (!result.authenticated) {
   962→        router.push('/merchant-os/login')
   963→      }
   964→      
   965→      setIsLoading(false)
   966→    }
   967→    init()
   968→  }, [router])
   969→
   970→  if (isLoading) {
   971→    return (
   972→      <div className="min-h-screen flex items-center justify-center bg-[#FAFAFA]">

   981→  if (!user) {

   985→  return (
