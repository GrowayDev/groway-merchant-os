   270→const formatCurrency = (amount: number) => {
   271→  return `R${Math.abs(amount).toLocaleString('en-ZA', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`
   272→}
   273→
   274→const formatPercent = (value: number) => {
   275→  return `${value >= 0 ? '+' : ''}${value.toFixed(1)}%`
   276→}
   277→
   278→const getDaysUntil = (dateStr: string) => {
   279→  const date = new Date(dateStr)
   280→  const now = new Date()
   281→  const diff = Math.ceil((date.getTime() - now.getTime()) / (1000 * 60 * 60 * 24))
   282→  return diff
   283→}

   289→export function GroCreditView() {

   291→  const { merchantId, isLoading: isLoadingMerchant } = useMerchantData()

   294→  const [activeTab, setActiveTab] = useState<string>('overview')
   295→  const [loading, setLoading] = useState(true)
   296→  const [processing, setProcessing] = useState(false)
   297→

   299→  const [salesPerformance, setSalesPerformance] = useState<SalesPerformance | null>(null)
   300→  const [inventoryIntelligence, setInventoryIntelligence] = useState<InventoryIntelligence | null>(null)
   301→  const [supplyIntelligence, setSupplyIntelligence] = useState<SupplyIntelligence | null>(null)
   302→  const [creditScore, setCreditScore] = useState<CreditScoreBreakdown | null>(null)
   303→  const [facility, setFacility] = useState<CreditFacility | null>(null)
   304→  const [drawTransactions, setDrawTransactions] = useState<DrawTransaction[]>([])
   305→  
   306→  // Dialog states
   307→  const [drawDialogOpen, setDrawDialogOpen] = useState(false)
   308→  const [paymentDialogOpen, setPaymentDialogOpen] = useState(false)
   309→  const [applyDialogOpen, setApplyDialogOpen] = useState(false)
   310→  const [algorithmDialogOpen, setAlgorithmDialogOpen] = useState(false)
   311→  const [groSupplyOrderDialogOpen, setGroSupplyOrderDialogOpen] = useState(false)
   312→  
   313→  // Form states
   314→  const [drawAmount, setDrawAmount] = useState<number>(0)
   315→  const [drawPurpose, setDrawPurpose] = useState<string>('working_capital')
   316→  const [paymentAmount, setPaymentAmount] = useState<number>(0)
   317→  const [applicationForm, setApplicationForm] = useState({
   318→    businessType: 'pty_ltd' as 'sole_proprietor' | 'pty_ltd' | 'cc' | 'other',
   319→    tradingYears: '',
   320→    requestedTier: 'business' as CreditTier,
   321→    purpose: '',
   322→    acceptNCA: false,
   323→    acceptTerms: false
   324→  })
   325→
   326→  // Fetch all data
   327→  const fetchData = useCallback(async () => {
   328→    if (!merchantId) return
   329→    setLoading(true)
   330→    try {
   331→      // Fetch from API endpoints
   332→      const [salesRes, inventoryRes, supplyRes, creditRes, facilityRes, transactionsRes] = await Promise.all([
   333→        fetch(`/merchant-os/api/grocredit/sales-performance?merchantId=${merchantId}`),
   334→        fetch(`/merchant-os/api/grocredit/inventory-intelligence?merchantId=${merchantId}`),
   335→        fetch(`/merchant-os/api/grocredit/supply-intelligence?merchantId=${merchantId}`),
   336→        fetch(`/merchant-os/api/grocredit/credit-score?merchantId=${merchantId}`),
   337→        fetch(`/merchant-os/api/grocredit/facility?merchantId=${merchantId}`),
   338→        fetch(`/merchant-os/api/grocredit/transactions?merchantId=${merchantId}`)
   339→      ])
   340→      
   341→      if (salesRes.ok) {
   342→        const data = await salesRes.json()
   343→        setSalesPerformance(data.salesPerformance || null)
   344→      }
   345→      
   346→      if (inventoryRes.ok) {
   347→        const data = await inventoryRes.json()
   348→        setInventoryIntelligence(data.inventoryIntelligence || null)
   349→      }
   350→      
   351→      if (supplyRes.ok) {
   352→        const data = await supplyRes.json()
   353→        setSupplyIntelligence(data.supplyIntelligence || null)
   354→      }
   355→      
   356→      if (creditRes.ok) {
   357→        const data = await creditRes.json()
   358→        setCreditScore(data.creditScore || null)
   359→      }
   360→      
   361→      if (facilityRes.ok) {
   362→        const data = await facilityRes.json()
   363→        setFacility(data.facility || null)
   364→      }
   365→      
   366→      if (transactionsRes.ok) {
   367→        const data = await transactionsRes.json()
   368→        setDrawTransactions(data.transactions || [])
   369→      }
   370→      
   371→    } catch (error) {
   372→      console.error('Error fetching GroCredit data:', error)
   373→      toast.error('Failed to load credit data')
   374→    } finally {
   375→      setLoading(false)
   376→    }
   377→  }, [merchantId])
   378→
   379→  useEffect(() => {
   380→    fetchData()
   381→  }, [fetchData])
   382→
   383→  // Calculate recommended credit limit based on algorithm
   384→  const recommendedLimit = useMemo(() => {
   385→    if (!salesPerformance || !creditScore) return 0
   386→    
   387→    // Algorithm: Based on monthly sales * credit score factor * tier multiplier
   388→    const baseLimit = salesPerformance.monthlyTotal * 0.6 // 60% of monthly sales as base
   389→    const scoreMultiplier = creditScore.overallScore / 100
   390→    const tierMultiplier = creditScore.tier === 'enterprise' ? 1.5 :
   391→                          creditScore.tier === 'business' ? 1.0 :
   392→                          creditScore.tier === 'seasonal' ? 0.8 : 0.5
   393→    
   394→    return Math.round(baseLimit * scoreMultiplier * tierMultiplier)
   395→  }, [salesPerformance, creditScore])
   396→
   397→  // Calculate interest for amount
   398→  const calculateInterest = (amount: number, months: number = 1) => {
   399→    if (!facility) return 0
   400→    const monthlyRate = facility.interestRate / 100 / 12
   401→    return amount * monthlyRate * months
   402→  }
   403→
   404→  // Handle draw
   405→  const handleDraw = async () => {
   406→    if (!facility || drawAmount <= 0 || drawAmount > facility.availableCredit) {
   407→      toast.error('Invalid draw amount')
   408→      return
   409→    }
   410→    
   411→    setProcessing(true)
   412→    try {
   413→      // Simulate API call
   414→      await new Promise(resolve => setTimeout(resolve, 1500))
   415→      
   416→      const newDraw: DrawTransaction = {
   417→        id: `DRW-${Date.now()}`,
   418→        reference: `GC-DRW-${new Date().getFullYear()}-${String(drawTransactions.length + 1).padStart(3, '0')}`,
   419→        amount: drawAmount,
   420→        purpose: drawPurpose as any,
   421→        status: 'processed',
   422→        createdAt: new Date().toISOString(),
   423→        interestAccrued: calculateInterest(drawAmount),
   424→        repaymentDate: `${new Date().getFullYear()}-${String(new Date().getMonth() + 2).padStart(2, '0')}-25`
   425→      }
   426→      
   427→      setDrawTransactions(prev => [newDraw, ...prev])
   428→      setFacility(prev => prev ? {
   429→        ...prev,
   430→        availableCredit: prev.availableCredit - drawAmount,
   431→        usedCredit: prev.usedCredit + drawAmount
   432→      } : null)
   433→      
   434→      setDrawDialogOpen(false)
   435→      setDrawAmount(0)
   436→      toast.success(`Successfully drew ${formatCurrency(drawAmount)}`)
   437→    } catch (error) {
   438→      toast.error('Failed to process draw')
   439→    } finally {
   440→      setProcessing(false)
   441→    }
   442→  }
   443→
   444→  // Handle payment
   445→  const handlePayment = async () => {
   446→    if (!facility || paymentAmount <= 0) {
   447→      toast.error('Invalid payment amount')
   448→      return
   449→    }
   450→    
   451→    setProcessing(true)
   452→    try {
   453→      await new Promise(resolve => setTimeout(resolve, 1500))
   454→      
   455→      const actualPayment = Math.min(paymentAmount, facility.usedCredit)
   456→      
   457→      setFacility(prev => prev ? {
   458→        ...prev,
   459→        availableCredit: prev.availableCredit + actualPayment,
   460→        usedCredit: prev.usedCredit - actualPayment
   461→      } : null)
   462→      
   463→      setPaymentDialogOpen(false)
   464→      setPaymentAmount(0)
   465→      toast.success(`Payment of ${formatCurrency(actualPayment)} processed successfully`)
   466→    } catch (error) {
   467→      toast.error('Failed to process payment')
   468→    } finally {
   469→      setProcessing(false)
   470→    }
   471→  }
   472→
   473→  // Handle GroSupply order payment
   474→  const handleGroSupplyPayment = async (poId: string, amount: number) => {
   475→    if (!facility || amount > facility.availableCredit) {
   476→      toast.error('Insufficient credit available')
   477→      return
   478→    }
   479→    
   480→    setProcessing(true)
   481→    try {
   482→      await new Promise(resolve => setTimeout(resolve, 1500))
   483→      
   484→      const newDraw: DrawTransaction = {
   485→        id: `DRW-${Date.now()}`,
   486→        reference: `GC-GS-${new Date().getFullYear()}-${String(drawTransactions.length + 1).padStart(3, '0')}`,
   487→        amount,
   488→        purpose: 'grosupply_order',
   489→        linkedPOId: poId,
   490→        status: 'processed',
   491→        createdAt: new Date().toISOString(),
   492→        interestAccrued: calculateInterest(amount),
   493→        repaymentDate: `${new Date().getFullYear()}-${String(new Date().getMonth() + 2).padStart(2, '0')}-25`
   494→      }
   495→      
   496→      setDrawTransactions(prev => [newDraw, ...prev])
   497→      setFacility(prev => prev ? {
   498→        ...prev,
   499→        availableCredit: prev.availableCredit - amount,
   500→        usedCredit: prev.usedCredit + amount
   501→      } : null)
   502→      
   503→      setGroSupplyOrderDialogOpen(false)
   504→      toast.success(`GroSupply order ${poId} paid via GroCredit`)
   505→    } catch (error) {
   506→      toast.error('Failed to process GroSupply payment')

   508→      setProcessing(false)
   509→    }
   510→  }

   528→        <div className="text-center">
   529→          <Loader2 className="h-8 w-8 animate-spin text-[#00897B] mx-auto mb-4" />
   530→          <p className="text-sm text-[#757575]">Analyzing your business performance...</p>
   531→        </div>
   532→      </div>
   533→    )
   534→  }
   535→
   536→  return (
   537→    <div className="flex-1 flex flex-col min-h-0">
   538→      {/* Header */}
   539→      <div className="p-4 border-b border-gray-200 bg-white">
   540→        <div className="flex items-center justify-between">
   541→          <div className="flex items-center gap-3">
   542→            <div className="w-10 h-10 rounded-lg bg-gradient-to-br from-violet-500 to-purple-600 flex items-center justify-center">
   543→              <PiggyBank className="h-5 w-5 text-white" />
   544→            </div>
   545→            <div>
   546→              <div className="flex items-center gap-2">
   547→                <h2 className="text-lg font-bold text-[#1A1A1A]">GroCredit</h2>
   548→                <Badge className="bg-gradient-to-r from-violet-500 to-purple-500 text-white text-[10px]">
   549→                  <Sparkles className="h-3 w-3 mr-1" />
   550→                  WORKING CAPITAL
   551→                </Badge>
   552→              </div>
   553→              <p className="text-xs text-[#757575]">Algorithm-Based Merchant Financing</p>
   554→            </div>
   555→          </div>
   556→          <div className="flex items-center gap-2">
   557→            {facility && <TierBadge tier={facility.tier} />}
   558→            <Button
   559→              variant="outline"
   560→              size="sm"
   561→              onClick={() => setAlgorithmDialogOpen(true)}
   562→            >
   563→              <Brain className="h-4 w-4 mr-1" />
   564→              How It Works
   565→            </Button>
   566→            {facility?.status === 'active' && (
   567→              <Button
   568→                className="bg-gradient-to-r from-violet-500 to-purple-500 hover:from-violet-600 hover:to-purple-600"
   569→                size="sm"
   570→                onClick={() => setDrawDialogOpen(true)}
   571→              >
   572→                <Plus className="h-4 w-4 mr-1" />
   573→                Draw Funds
   574→              </Button>
   575→            )}
   576→          </div>
   577→        </div>
   578→      </div>
   579→
   580→      {/* Tabs */}
   581→      <div className="px-4 pt-3 bg-white border-b border-gray-200">
   582→        <div className="flex gap-1 overflow-x-auto">
   583→          {[
   584→            { id: 'overview', label: 'Overview', icon: Wallet },
   585→            { id: 'algorithm', label: 'Credit Algorithm', icon: Brain },
