   321→interface DiscountDialogProps {
   322→  open: boolean
   323→  onOpenChange: (open: boolean) => void
   324→  mode: 'item' | 'cart'

   329→
   330→export function DiscountDialog({ open, onOpenChange, mode, item, cartTotal, onApply }: DiscountDialogProps) {
   331→  const [discountType, setDiscountType] = useState<'percentage' | 'fixed'>('percentage')
   332→  const [discountValue, setDiscountValue] = useState('')
   333→  const [reason, setReason] = useState('')
   334→
   335→  const quickDiscounts = [
   336→    { type: 'percentage' as const, value: 5, label: '5%' },
   337→    { type: 'percentage' as const, value: 10, label: '10%' },
   338→    { type: 'percentage' as const, value: 15, label: '15%' },
   339→    { type: 'percentage' as const, value: 20, label: '20%' },
   340→    { type: 'percentage' as const, value: 25, label: '25%' },
   341→    { type: 'percentage' as const, value: 50, label: '50%' },
   342→  ]
   343→
   344→  // Reset state when dialog closes using the dialog's onOpenChange
   345→  const handleOpenChange = (isOpen: boolean) => {
   346→    if (!isOpen) {
   347→      setDiscountValue('')
   348→      setReason('')
   349→      setDiscountType('percentage')
   350→    }
   351→    onOpenChange(isOpen)
   352→  }
   353→
   354→  const handleApply = () => {
   355→    const value = parseFloat(discountValue)
   356→    if (!isNaN(value) && value > 0) {
   357→      onApply({ type: discountType, value, reason }, item?.id)
   358→      onOpenChange(false)
   359→    }
   360→  }
   361→
   362→  const handleQuickDiscount = (type: 'percentage' | 'fixed', value: number) => {
   363→    onApply({ type, value, reason: 'Quick discount' }, item?.id)
   364→    onOpenChange(false)
   365→  }
   366→
   367→  const baseAmount = mode === 'item' ? (item?.price || 0) * (item?.quantity || 1) : (cartTotal || 0)
   368→
   369→  return (
   370→    <Dialog open={open} onOpenChange={onOpenChange}>
   371→      <DialogContent className="sm:max-w-md">
   372→        <DialogHeader>
   373→          <DialogTitle className="flex items-center gap-2">
   374→            <Percent className="h-5 w-5 text-[#00897B]" />
   375→            {mode === 'item' ? 'Item Discount' : 'Cart Discount'}
   376→          </DialogTitle>
   377→          <DialogDescription>
   378→            {mode === 'item' && item
   379→              ? `Apply discount to "${item.name}"`
   380→              : `Apply discount to entire cart`}
   381→          </DialogDescription>
   382→        </DialogHeader>
   383→        <div className="space-y-4 py-4">
   384→          <div className="bg-gray-50 rounded-lg p-3">
   385→            <p className="text-sm text-gray-500">{mode === 'item' ? 'Item Total' : 'Cart Total'}</p>
   386→            <p className="text-lg font-bold text-[#1A1A1A]">R{baseAmount.toFixed(2)}</p>
   387→          </div>
   388→
   389→          <div>
   390→            <Label className="text-sm mb-2 block">Quick Discounts</Label>
   391→            <div className="grid grid-cols-6 gap-2">
   392→              {quickDiscounts.map((d) => (
   393→                <Button
   394→                  key={d.value}
   395→                  variant="outline"
   396→                  size="sm"
   397→                  className="h-9"
   398→                  onClick={() => handleQuickDiscount(d.type, d.value)}
   399→                >
   400→                  {d.label}
   401→                </Button>
   402→              ))}
   403→            </div>
   404→          </div>
   405→
   406→          <Separator />
   407→
   408→          <div className="space-y-3">
   409→            <Label>Custom Discount</Label>
   410→            <div className="flex gap-2">
   411→              <Button
   412→                variant={discountType === 'percentage' ? 'default' : 'outline'}
   413→                size="sm"
   414→                className={discountType === 'percentage' ? 'bg-[#00897B] hover:bg-[#00695C]' : ''}
   415→                onClick={() => setDiscountType('percentage')}
   416→              >
   417→                <Percent className="h-3 w-3 mr-1" />
   418→                Percentage
   419→              </Button>
   420→              <Button
   421→                variant={discountType === 'fixed' ? 'default' : 'outline'}
   422→                size="sm"
   423→                className={discountType === 'fixed' ? 'bg-[#00897B] hover:bg-[#00695C]' : ''}
   424→                onClick={() => setDiscountType('fixed')}
   425→              >
   426→                <DollarSign className="h-3 w-3 mr-1" />
   427→                Fixed Amount
   428→              </Button>
   429→            </div>
   430→            <div className="relative">
   431→              {discountType === 'fixed' && (
   432→                <span className="absolute left-3 top-1/2 -translate-y-1/2 text-sm text-gray-500">R</span>
   433→              )}
   434→              <Input
   435→                type="number"
   436→                value={discountValue}
   437→                onChange={(e) => setDiscountValue(e.target.value)}
   438→                placeholder={discountType === 'percentage' ? 'Enter %' : '0.00'}
   439→                className={discountType === 'fixed' ? 'pl-7' : ''}
   440→              />
   441→              {discountType === 'percentage' && (
   442→                <span className="absolute right-3 top-1/2 -translate-y-1/2 text-sm text-gray-500">%</span>
   443→              )}
   444→            </div>
   445→            {discountValue && (
   446→              <p className="text-sm text-[#00897B]">
   447→                Discount: -R{discountType === 'percentage'
   448→                  ? (baseAmount * parseFloat(discountValue) / 100).toFixed(2)
   449→                  : parseFloat(discountValue).toFixed(2)}
   450→              </p>
   451→            )}
   452→          </div>
   453→
   454→          <Textarea
   455→            value={reason}
   456→            onChange={(e) => setReason(e.target.value)}
   457→            placeholder="Reason for discount (optional)"
   458→            rows={2}
   459→          />
   460→        </div>
   461→        <DialogFooter>
   462→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   463→            Cancel
   464→          </Button>
   465→          <Button onClick={handleApply} className="bg-[#00897B] hover:bg-[#00695C]">
   466→            Apply Discount
   467→          </Button>
   468→        </DialogFooter>
   469→      </DialogContent>
   470→    </Dialog>
   471→  )
   472→}

   474→// Split Payment Dialog
   475→interface SplitPaymentProps {
   476→  open: boolean
   477→  onOpenChange: (open: boolean) => void
   478→  total: number
   479→  onComplete: (payments: SplitPayment[]) => void
   480→}
   481→
   482→export function SplitPaymentDialog({ open, onOpenChange, total, onComplete }: SplitPaymentProps) {
   483→  const [payments, setPayments] = useState<SplitPayment[]>([])
   484→  const [selectedMethod, setSelectedMethod] = useState<SplitPayment['method']>('cash')
   485→  const [amount, setAmount] = useState('')
   486→
   487→  const paymentMethods = [
   488→    { id: 'cash', icon: Banknote, label: 'Cash' },
   489→    { id: 'card', icon: CreditCard, label: 'Card' },
   490→    { id: 'qr', icon: QrCode, label: 'QR Pay' },
   491→    { id: 'wallet', icon: Wallet, label: 'Wallet' },
   492→    { id: 'voucher', icon: Tag, label: 'Voucher' },
   493→    { id: 'account', icon: Building2, label: 'On Account' },
   494→  ]
   495→
   496→  const totalPaid = payments.reduce((sum, p) => sum + p.amount, 0)
   497→  const remaining = total - totalPaid
   498→
   499→  // Handle open change with reset
   500→  const handleOpenChange = (isOpen: boolean) => {
   501→    if (!isOpen) {
   502→      setPayments([])
   503→      setAmount('')
   504→      setSelectedMethod('cash')
   505→    }
   506→    onOpenChange(isOpen)
   507→  }
   508→
   509→  const addPayment = () => {
   510→    const paymentAmount = parseFloat(amount)
   511→    if (!isNaN(paymentAmount) && paymentAmount > 0 && paymentAmount <= remaining) {
   512→      setPayments([
   513→        ...payments,
   514→        { id: Date.now().toString(), method: selectedMethod, amount: paymentAmount }
   515→      ])
   516→      setAmount('')
   517→    }
   518→  }
   519→
   520→  const removePayment = (id: string) => {
   521→    setPayments(payments.filter((p) => p.id !== id))
   522→  }
   523→
   524→  const addExactRemaining = () => {
   525→    if (remaining > 0) {
   526→      setAmount(remaining.toFixed(2))
   527→    }
   528→  }
   529→
   530→  const handleComplete = () => {
   531→    if (remaining === 0 && payments.length > 0) {
   532→      onComplete(payments)
   533→      onOpenChange(false)
   534→    }
   535→  }
   536→
   537→  return (
   538→    <Dialog open={open} onOpenChange={handleOpenChange}>
   539→      <DialogContent className="sm:max-w-lg">
   540→        <DialogHeader>
   541→          <DialogTitle className="flex items-center gap-2">
   542→            <ArrowRightLeft className="h-5 w-5 text-[#00897B]" />
   543→            Split Payment
   544→          </DialogTitle>
   545→          <DialogDescription>
   546→            Total amount: R{total.toFixed(2)}
   547→          </DialogDescription>
   548→        </DialogHeader>
   549→        <div className="space-y-4 py-4">
   550→          {/* Payment Progress */}
   551→          <div className="bg-gray-50 rounded-lg p-4">
   552→            <div className="flex justify-between mb-2">
   553→              <span className="text-sm text-gray-500">Total</span>
   554→              <span className="font-bold">R{total.toFixed(2)}</span>
   555→            </div>
   556→            <div className="flex justify-between mb-2">
   557→              <span className="text-sm text-gray-500">Paid</span>
   558→              <span className="font-medium text-green-600">R{totalPaid.toFixed(2)}</span>
   559→            </div>
   560→            <Separator className="my-2" />
   561→            <div className="flex justify-between">
   562→              <span className="text-sm text-gray-500">Remaining</span>
   563→              <span className={`font-bold ${remaining > 0 ? 'text-amber-600' : 'text-green-600'}`}>
   564→                R{remaining.toFixed(2)}
   565→              </span>
   566→            </div>
   567→          </div>
   568→
   569→          {/* Added Payments */}
   570→          {payments.length > 0 && (
   571→            <div className="space-y-2">
   572→              <Label className="text-sm">Payments Added</Label>
   573→              <div className="space-y-2">
   574→                {payments.map((payment) => (
   575→                  <div key={payment.id} className="flex items-center justify-between bg-[#E0F2F1] rounded-lg p-3">
   576→                    <div className="flex items-center gap-2">
   577→                      {paymentMethods.find((m) => m.id === payment.method)?.icon && (
   578→                        <div className="w-8 h-8 rounded-full bg-white flex items-center justify-center">
   579→                          {(() => {
   580→                            const Icon = paymentMethods.find((m) => m.id === payment.method)!.icon
   581→                            return <Icon className="h-4 w-4 text-[#00897B]" />
   582→                          })()}
   583→                        </div>
   584→                      )}
   585→                      <span className="font-medium capitalize">{payment.method}</span>
   586→                    </div>
   587→                    <div className="flex items-center gap-3">
   588→                      <span className="font-bold">R{payment.amount.toFixed(2)}</span>
   589→                      <Button
   590→                        variant="ghost"
   591→                        size="sm"
   592→                        className="h-6 w-6 p-0 text-red-500"
   593→                        onClick={() => removePayment(payment.id)}
   594→                      >
   595→                        <X className="h-4 w-4" />
   596→                      </Button>
   597→                    </div>
   598→                  </div>
   599→                ))}
   600→              </div>
   601→            </div>
   602→          )}
   603→
   604→          {/* Add Payment */}
   605→          {remaining > 0 && (
   606→            <div className="space-y-3 border-t pt-4">
   607→              <Label className="text-sm">Add Payment</Label>
   608→              <div className="grid grid-cols-3 gap-2">
   609→                {paymentMethods.map((method) => (
