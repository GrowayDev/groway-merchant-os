    17→    $allModels: {
    18→      async $allOperations({ args, query }) {
    19→        const fill = (row: unknown) => {
    20→          if (row && typeof row === 'object' && !Array.isArray(row) && (row as Record<string, unknown>).updatedAt === undefined) {
    21→            (row as Record<string, unknown>).updatedAt = new Date();
    22→          }
    23→        };
    24→        const d = (args as Record<string, unknown> | undefined)?.data;
    25→        if (d !== undefined) {
    26→          if (Array.isArray(d)) d.forEach(fill); else fill(d);
    27→        }
    28→        return query(args as never);
    29→      },
    30→    },
    31→  },
