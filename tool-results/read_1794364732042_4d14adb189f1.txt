   375→   375→        item,
   376→   376→        type: 'low_stock',

   379→   379→        acknowledged: false
   380→   380→      })
   381→   381→    }

   383→   383→    // Check for expiring batches

   385→   385→      item.batches.forEach(batch => {
   386→   386→        if (batch.status === 'expiring_soon') {
   387→   387→          alerts.push({
   388→   388→            id: `ALERT-${item.id}-${batch.batchNo}`,
   389→   389→            item,
   390→   390→            type: 'expiring_soon',
   391→   391→            message: `${item.name} batch ${batch.batchNo} expires on ${batch.expiry}`,
   392→   392→            createdAt: new Date().toISOString(),
   393→   393→            acknowledged: false
   394→   394→          })
   395→   395→        }
   396→   396→      })
   397→   397→    }
   398→   398→  })
   399→   399→  
   400→   400→  return alerts
   401→   401→}
