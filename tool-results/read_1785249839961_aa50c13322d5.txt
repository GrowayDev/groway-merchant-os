   286→export function CourseManagementView() {
   287→  const [orders, setOrders] = useState<TableOrder[]>([])
   288→  const [selectedOrder, setSelectedOrder] = useState<TableOrder | null>(null)
   289→  const [currentTime, setCurrentTime] = useState(new Date())
   290→  const [fireCourseDialogOpen, setFireCourseDialogOpen] = useState(false)
   291→  const [fireItemDialogOpen, setFireItemDialogOpen] = useState(false)
   292→  const [holdCourseDialogOpen, setHoldCourseDialogOpen] = useState(false)
   293→  const [rushDialogOpen, setRushDialogOpen] = useState(false)
   294→  const [messageDialogOpen, setMessageDialogOpen] = useState(false)
   295→  const [selectedCourseToFire, setSelectedCourseToFire] = useState<Course | null>(null)
   296→  const [selectedItemToFire, setSelectedItemToFire] = useState<{ course: Course; item: CourseItem } | null>(null)
   297→  const [selectedCourseToHold, setSelectedCourseToHold] = useState<Course | null>(null)
   298→  const [loading, setLoading] = useState(true)
   299→  const [refreshing, setRefreshing] = useState(false)
   300→  const [newMessage, setNewMessage] = useState('')
   301→  const [holdReason, setHoldReason] = useState('')
   302→  const [holdMinutes, setHoldMinutes] = useState(10)
   303→  const [rushReason, setRushReason] = useState('')
   304→  const [selectedItemIds, setSelectedItemIds] = useState<Set<string>>(new Set())
   305→
   306→  // Fetch orders from API
   307→  const fetchOrders = useCallback(async () => {
   308→    try {
   309→      const [posRes, kitchenRes] = await Promise.all([
   310→        fetch('/merchant-os/api/pos/orders?status=open'),
   311→        fetch('/merchant-os/api/kitchen/orders')
   312→      ])
   313→      
   314→      const posData = await posRes.json()
   315→      const kitchenData = await kitchenRes.json()
   316→      
   317→      // Transform POS orders to course format
   318→      const transformedOrders = (posData.data || []).map((order: any) => {
   319→        // Group items by course type based on category/station
   320→        const courseGroups: Record<string, CourseItem[]> = {}
   321→        
   322→        order.items?.forEach((item: any) => {
   323→          const courseType = getCourseTypeFromItem(item)
   324→          if (!courseGroups[courseType]) {
   325→            courseGroups[courseType] = []
   326→          }
   327→          courseGroups[courseType].push({
   328→            id: item.id,
   329→            name: item.name,
   330→            quantity: item.quantity,
   331→            price: item.unitPrice,
   332→            notes: item.notes,
   333→            station: item.station || 'grill',
   334→            status: mapKitchenStatus(item.status || 'pending'),
   335→            preparationTime: item.preparationTime || 15,
   336→            dietary: item.dietary ? JSON.parse(item.dietary) : [],
   337→          })
   338→        })
   339→        
   340→        // Convert to courses array
   341→        const courses: Course[] = Object.entries(courseGroups).map(([type, items], index) => {
   342→          const config = courseConfig[type as CourseType] || courseConfig.main
   343→          return {
   344→            id: `course-${order.id}-${index}`,
   345→            type: type as CourseType,
   346→            name: config.name,
   347→            items,
   348→            status: getCourseStatusFromItems(items),
   349→            orderedAt: order.createdAt ? new Date(order.createdAt) : undefined,
   350→            estimatedTime: Math.max(...items.map(i => i.preparationTime), 15),
   351→            priority: 'normal' as const,
   352→          }
   353→        }).sort((a, b) => courseConfig[a.type].order - courseConfig[b.type].order)
   354→        
   355→        return {
   356→          id: order.id,
   357→          tableNumber: order.tableNumber || '-',
   358→          guests: order.guests || 1,
   359→          serverName: order.staffName || order.staff?.name || 'Unassigned',
   360→          courses,
   361→          status: 'active' as const,
   362→          startedAt: new Date(order.createdAt),
   363→          messages: [],
   364→        }
   365→      })
   366→      
   367→      // If no real orders, use sample data
   368→      if (transformedOrders.length === 0) {
   369→        setOrders(sampleOrders)
   370→      } else {
   371→        setOrders(transformedOrders)
   372→      }
   373→    } catch (error) {
   374→      console.error('Error fetching orders:', error)
   375→      // Use sample data on error
   376→      setOrders(sampleOrders)
   377→    } finally {
   378→      setLoading(false)
   379→      setRefreshing(false)
   380→    }
   381→  }, [])
   382→
   383→  // Helper to determine course type from item
   384→  const getCourseTypeFromItem = (item: any): CourseType => {
   385→    const station = item.station?.toLowerCase() || ''
   386→    const category = (item.category || '').toLowerCase()
   387→    
   388→    if (station.includes('bar') || category.includes('drink') || category.includes('beverage')) return 'drinks'
   389→    if (station.includes('dessert') || category.includes('dessert')) return 'dessert'
   390→    if (category.includes('starter') || category.includes('appetizer') || category.includes('start')) return 'starter'
   391→    if (category.includes('side')) return 'sides'
   392→    return 'main'
   393→  }
   394→
   395→  // Map kitchen status to course status
   396→  const mapKitchenStatus = (status: string): CourseStatus => {
   397→    const statusMap: Record<string, CourseStatus> = {
   398→      'pending': 'pending',
   399→      'held': 'held',
   400→      'ordered': 'ordered',
   401→      'preparing': 'preparing',
   402→      'ready': 'ready',
   403→      'served': 'served',
   404→      'completed': 'served',
   405→    }
   406→    return statusMap[status.toLowerCase()] || 'pending'
   407→  }
   408→
   409→  // Get course status from items
   410→  const getCourseStatusFromItems = (items: CourseItem[]): CourseStatus => {
   411→    if (items.every(i => i.status === 'served')) return 'served'
   412→    if (items.every(i => i.status === 'ready')) return 'ready'
   413→    if (items.some(i => i.status === 'preparing')) return 'preparing'
   414→    if (items.some(i => i.status === 'ordered')) return 'ordered'
   415→    if (items.some(i => i.status === 'held')) return 'held'
   416→    return 'pending'
   417→  }
   418→
   419→  // Fetch orders on mount and poll every 10 seconds
   420→  useEffect(() => {
   421→    fetchOrders()
   422→    const interval = setInterval(fetchOrders, 10000)
   423→    return () => clearInterval(interval)
   424→  }, [fetchOrders])
   425→
   426→  // Update time every second
   427→  useEffect(() => {
   428→    const interval = setInterval(() => setCurrentTime(new Date()), 1000)
   429→    return () => clearInterval(interval)
   430→  }, [])
   431→
   432→  // Refresh handler
   433→  const handleRefresh = async () => {
   434→    setRefreshing(true)
   435→    await fetchOrders()
   436→    toast.success('Orders refreshed')
   437→  }
   438→
   439→  // Calculate time elapsed
   440→  const getTimeElapsed = (date: Date) => {
   441→    const elapsed = Math.floor((currentTime.getTime() - date.getTime()) / 60000)
   442→    return elapsed
   443→  }
   444→
   445→  // Calculate time remaining for course
   446→  const getTimeRemaining = (course: Course) => {
   447→    if (!course.orderedAt) return null
   448→    const elapsed = Math.floor((currentTime.getTime() - course.orderedAt.getTime()) / 60000)
   449→    return course.estimatedTime - elapsed
   450→  }
   451→
   452→  // Get current active course for a table
   453→  const getActiveCourse = (order: TableOrder): Course | null => {
   454→    // First check for ready courses (priority)
   455→    const readyCourse = order.courses.find(c => c.status === 'ready')
   456→    if (readyCourse) return readyCourse
   457→    
   458→    // Then check for preparing courses
   459→    const preparingCourse = order.courses.find(c => c.status === 'preparing')
   460→    if (preparingCourse) return preparingCourse
   461→    
   462→    // Then check for ordered courses
   463→    const orderedCourse = order.courses.find(c => c.status === 'ordered')
   464→    if (orderedCourse) return orderedCourse
   465→    
   466→    // Then check for held courses
   467→    const heldCourse = order.courses.find(c => c.status === 'held')
   468→    if (heldCourse) return heldCourse
   469→    
   470→    // Then check for pending courses (in order)
   471→    const pendingCourses = order.courses.filter(c => c.status === 'pending')
   472→    if (pendingCourses.length > 0) {
   473→      return pendingCourses.sort((a, b) => courseConfig[a.type].order - courseConfig[b.type].order)[0]
   474→    }
   475→    
   476→    return null
   477→  }
   478→
   479→  // Check if course timing is overdue
   480→  const isCourseOverdue = (course: Course): { overdue: boolean; minutes: number } => {
   481→    if (!course.orderedAt || course.status === 'served' || course.status === 'ready') {
   482→      return { overdue: false, minutes: 0 }
   483→    }
   484→    const elapsed = Math.floor((currentTime.getTime() - course.orderedAt.getTime()) / 60000)
   485→    const overdue = elapsed > course.estimatedTime
   486→    return { overdue, minutes: Math.max(0, elapsed - course.estimatedTime) }
   487→  }
   488→
   489→  // Fire course (send to kitchen)
   490→  const fireCourse = (orderId: string, courseId: string) => {
   491→    setOrders(prev => prev.map(order => {
   492→      if (order.id === orderId) {
   493→        return {
   494→          ...order,
   495→          courses: order.courses.map(course => {
   496→            if (course.id === courseId) {
   497→              return {
   498→                ...course,
   499→                status: 'ordered' as CourseStatus,
   500→                orderedAt: new Date(),
   501→                items: course.items.map(item => ({ 
   502→                  ...item, 
   503→                  status: 'ordered' as CourseStatus,
   504→                  firedAt: new Date()
