    70→```ts
    71→import { LedgerService } from '../ledger/ledger';
    72→import { mulBps } from '../shared/money';
    73→
    74→/**
    75→ * Funding is a mesh of facilities, not one lender. Each facility has its own currency, scope, limit, price and
    76→ * performance triggers, all held as data, so a new funder (bank, development lender, pension fund, impact
    77→ * fund or a digital-native source) is added by configuration. Funding is matched to the currency of the asset,
    78→ * so a kwacha advance is never funded from a rand facility.
    79→ */
    80→export interface Facility {
    81→  id: string; funder: string; countries: ('ZA' | 'MW')[]; currency: string; products: string[] | '*';
    82→  limitMinor: bigint; drawnMinor: bigint;
    83→  /** Lower number is used first. */
    84→  priority: number;
    85→  /** Funder's cost on each advance, in basis points of principal (a flat simplification). */
    86→  costBps: number;
    87→  /** Performance triggers: breaching either suspends new draws. */
    88→  minRecoveryRateBps?: number; maxChargeOffBps?: number;
    89→  status: 'active' | 'suspended' | 'amortising';
    90→}
    91→export interface Allocation { requestId: string; facilityId: string; principalMinor: bigint; costBps: number; repaidMinor: bigint }
    92→
    93→export interface FacilityRepo {
    94→  list(): Promise<Facility[]>;
    95→  save(f: Facility): Promise<void>;
    96→  allocation(requestId: string): Promise<Allocation | undefined>;
    97→  saveAllocation(a: Allocation): Promise<void>;
    98→}
    99→export class InMemoryFacilityRepo implements FacilityRepo {
   100→  private f = new Map<string, Facility>(); private a = new Map<string, Allocation>();
   101→  constructor(facilities: Facility[] = []) { facilities.forEach((x) => this.f.set(x.id, { ...x })); }
   102→  async list() { return [...this.f.values()]; }
   103→  async save(x: Facility) { this.f.set(x.id, x); }
   104→  async allocation(r: string) { return this.a.get(r); }
   105→  async saveAllocation(x: Allocation) { this.a.set(x.requestId, x); }
   106→}
   107→
   108→export class FundingError extends Error { constructor(public readonly code: string, m: string) { super(m); } }
   109→
   110→export const fundingAcct = {
   111→  supplier: (commodity: string) => `supplier_payable:${commodity}`,
   112→  funder: (facilityId: string) => `funder_payable:${facilityId}`,
   113→  cost: 'funding_cost_expense',
   114→  clearing: 'collections_clearing',
   115→};
   116→
   117→/** Peak float needed: daily advances times the days they stay outstanding, plus a safety buffer. */
   118→export function estimateFloat(p: { dailyAdvancesMinor: bigint; avgDaysOutstanding: number; bufferBps: number }): bigint {
   119→  if (p.avgDaysOutstanding < 0 || p.bufferBps < 0) throw new RangeError('inputs must be non-negative');
   120→  return (p.dailyAdvancesMinor * BigInt(Math.ceil(p.avgDaysOutstanding)) * BigInt(10000 + p.bufferBps)) / 10000n;
   121→}
   122→
   123→export class FundingService {
   124→  constructor(private ledger: LedgerService, private repo: FacilityRepo) {}
   125→
   126→  /** Choose a facility for an advance and move the obligation from the supplier to the funder. Idempotent per request. */
   127→  async allocate(a: { tenantId: string; requestId: string; productId: string; country: 'ZA' | 'MW'; currency: string; commodity: string; principalMinor: bigint }) {
   128→    const prior = await this.repo.allocation(a.requestId);
   129→    if (prior) return { facilityId: prior.facilityId, replayed: true };
   130→    const eligible = (await this.repo.list())
   131→      .filter((f) => f.status === 'active' && f.currency === a.currency && f.countries.includes(a.country)
   132→        && (f.products === '*' || f.products.includes(a.productId)) && f.limitMinor - f.drawnMinor >= a.principalMinor)
   133→      .sort((x, y) => x.priority - y.priority || Number((x.drawnMinor * 10000n) / (x.limitMinor || 1n) - (y.drawnMinor * 10000n) / (y.limitMinor || 1n)));
   134→    const f = eligible[0];
   135→    if (!f) throw new FundingError('FUNDING_UNAVAILABLE', 'no facility can fund this advance');
   136→    await this.ledger.post({
   137→      tenantId: a.tenantId, idempotencyKey: `${a.requestId}:funded`, reason: 'advance_funded', correlationId: a.requestId, currency: a.currency,
   138→      lines: [{ account: fundingAcct.supplier(a.commodity), amount: a.principalMinor }, { account: fundingAcct.funder(f.id), amount: -a.principalMinor }],
   139→    });
   140→    f.drawnMinor += a.principalMinor; await this.repo.save(f);
   141→    await this.repo.saveAllocation({ requestId: a.requestId, facilityId: f.id, principalMinor: a.principalMinor, costBps: f.costBps, repaidMinor: 0n });
   142→    return { facilityId: f.id, replayed: false };
   143→  }
   144→
   145→  /** Funding fell through (delivery failed): release the draw. */
   146→  async release(a: { tenantId: string; requestId: string; currency: string; commodity: string }) {
   147→    const al = await this.repo.allocation(a.requestId);
   148→    if (!al || al.repaidMinor > 0n) return;
   149→    await this.ledger.post({
   150→      tenantId: a.tenantId, idempotencyKey: `${a.requestId}:funding_released`, reason: 'advance_funding_released', correlationId: a.requestId, currency: a.currency,
   151→      lines: [{ account: fundingAcct.funder(al.facilityId), amount: al.principalMinor }, { account: fundingAcct.supplier(a.commodity), amount: -al.principalMinor }],
   152→    });
   153→    const f = (await this.repo.list()).find((x) => x.id === al.facilityId)!;
   154→    f.drawnMinor -= al.principalMinor; await this.repo.save(f);
   155→    await this.repo.saveAllocation({ ...al, repaidMinor: al.principalMinor });
   156→  }
   157→
   158→  /** Pay the funder its principal share, plus its cost, out of what was recovered from the customer. */
   159→  async repayFromRecovery(a: { tenantId: string; requestId: string; topupRef: string; currency: string; takenMinor: bigint; principalMinor: bigint; feeMinor: bigint }) {
   160→    const al = await this.repo.allocation(a.requestId);
   161→    if (!al) return { paid: 0n };
   162→    const share = (a.takenMinor * a.principalMinor) / (a.principalMinor + a.feeMinor);
   163→    const principalPart = share > al.principalMinor - al.repaidMinor ? al.principalMinor - al.repaidMinor : share;
   164→    if (principalPart <= 0n) return { paid: 0n };
   165→    const cost = mulBps(principalPart, BigInt(al.costBps));
   166→    const lines = [{ account: fundingAcct.funder(al.facilityId), amount: principalPart }, { account: fundingAcct.clearing, amount: -(principalPart + cost) }];
   167→    if (cost > 0n) lines.splice(1, 0, { account: fundingAcct.cost, amount: cost });
   168→    await this.ledger.post({ tenantId: a.tenantId, idempotencyKey: `${a.topupRef}:funder`, reason: 'funder_repaid', correlationId: a.topupRef, currency: a.currency, lines });
   169→    const f = (await this.repo.list()).find((x) => x.id === al.facilityId)!;
   170→    f.drawnMinor -= principalPart; await this.repo.save(f);
   171→    await this.repo.saveAllocation({ ...al, repaidMinor: al.repaidMinor + principalPart });
   172→    return { paid: principalPart + cost };
   173→  }
   174→
   175→  /** Performance triggers as code: suspend new draws on a breach, resume when performance recovers. */
   176→  async applyPerformance(stats: { recoveryRateBps: number; chargeOffBps: number }) {
   177→    for (const f of await this.repo.list()) {
   178→      if (f.status === 'amortising') continue;
   179→      const breach = (f.minRecoveryRateBps !== undefined && stats.recoveryRateBps < f.minRecoveryRateBps)
   180→        || (f.maxChargeOffBps !== undefined && stats.chargeOffBps > f.maxChargeOffBps);
   181→      const next = breach ? 'suspended' : 'active';
   182→      if (next !== f.status) { f.status = next; await this.repo.save(f); }
   183→    }
   184→  }
   185→
   186→  async headroom(currency: string) {
   187→    return (await this.repo.list()).filter((f) => f.status === 'active' && f.currency === currency).reduce((s, f) => s + (f.limitMinor - f.drawnMinor), 0n);
   188→  }
   189→}

   194→```ts
   195→export type Purpose = 'credit_underwriting' | 'wifi_billing' | 'marketing';
   196→interface ConsentEvent { subject: string; purpose: Purpose; kind: 'grant' | 'revoke'; at: Date; expiresAt?: Date }
   197→
   198→/** Append-only consent log. Latest event per (subject, purpose) wins. */
   199→export class ConsentService {
   200→  private log: ConsentEvent[] = [];
   201→  grant(subject: string, purpose: Purpose, at = new Date(), expiresAt?: Date) {
   202→    this.log.push({ subject, purpose, kind: 'grant', at, expiresAt });
   203→  }
   204→  revoke(subject: string, purpose: Purpose, at = new Date()) {
   205→    this.log.push({ subject, purpose, kind: 'revoke', at });
   206→  }
   207→  isAllowed(subject: string, purpose: Purpose, now = new Date()): boolean {
   208→    for (let i = this.log.length - 1; i >= 0; i--) {
   209→      const e = this.log[i]!;
   210→      if (e.subject !== subject || e.purpose !== purpose || e.at > now) continue;
   211→      if (e.kind === 'revoke') return false;
   212→      return !e.expiresAt || e.expiresAt > now;
   213→    }
   214→    return false;
   215→  }
   216→}

   221→```ts
   222→import { ConsentService } from '../consent/consent';
   223→import { fmtRand, mulBps, minOf } from '../shared/money';
   224→
   225→export interface FeatureStore { get(subject: string, names: string[]): Promise<Record<string, number | undefined>> }
   226→
   227→export interface Policy {
   228→  version: string;
   229→  modelVersion: string;
   230→  minTenureDays: number;
   231→  minLimit: bigint;
   232→  maxLimit: bigint;
   233→  /** Limit as a share of trailing 30-day sales, in basis points. */
   234→  limitShareOfSalesBps: number;
   235→}
   236→
   237→export type Outcome = 'approve' | 'decline' | 'refer';
   238→export interface TraceStep { step: string; result: 'pass' | 'fail' | 'skip'; detail: string }
   239→export interface Decision {
   240→  outcome: Outcome;
   241→  limit: bigint;
   242→  reasonCodes: string[];
   243→  /** Plain-language explanation and what would change the outcome. */
   244→  explanation: string[];
   245→  counterfactual: string[];
   246→  trace: TraceStep[];
   247→  versions: { policy: string; model: string };
   248→}
   249→
   250→const FEATURES = ['sales_30d_minor', 'tenure_days', 'arrears_minor', 'kyc_verified'] as const;
   251→
   252→export class DecisionOrchestrator {
   253→  constructor(private consent: ConsentService, private features: FeatureStore, private policy: Policy) {}
   254→
   255→  async decide(subject: string, now = new Date()): Promise<Decision> {
   256→    const trace: TraceStep[] = [];
   257→    const versions = { policy: this.policy.version, model: this.policy.modelVersion };
   258→    const out = (o: Outcome, limit: bigint, codes: string[], why: string[], cf: string[]): Decision =>
   259→      ({ outcome: o, limit, reasonCodes: codes, explanation: why, counterfactual: cf, trace, versions });
   260→
   261→    // 1. Consent is checked before any feature is read.
   262→    if (!this.consent.isAllowed(subject, 'credit_underwriting', now)) {
   263→      trace.push({ step: 'consent', result: 'fail', detail: 'no valid credit_underwriting consent' });
   264→      return out('refer', 0n, ['CONSENT_MISSING'], ['We need your permission to use your trading data for a credit check.'], ['Give consent for credit checks.']);
   265→    }
   266→    trace.push({ step: 'consent', result: 'pass', detail: 'credit_underwriting granted' });
   267→
   268→    // 2. Features. A missing critical feature means refer, never guess.
   269→    const f = await this.features.get(subject, [...FEATURES]);
   270→    const missing = FEATURES.filter((n) => f[n] === undefined);
   271→    if (missing.length) {
   272→      trace.push({ step: 'features', result: 'fail', detail: `missing: ${missing.join(', ')}` });
   273→      return out('refer', 0n, ['DATA_INCOMPLETE'], ['We do not yet have enough information to decide.'], ['A reviewer will check your details.']);
   274→    }
   275→    trace.push({ step: 'features', result: 'pass', detail: 'all critical features present' });
   276→    const sales = BigInt(f.sales_30d_minor!);
   277→    const tenure = f.tenure_days!;
   278→    const arrears = BigInt(f.arrears_minor!);
   279→
   280→    // 3. Hard stops.
   281→    if (f.kyc_verified !== 1) {
   282→      trace.push({ step: 'hard_stops', result: 'fail', detail: 'KYC not verified' });
   283→      return out('refer', 0n, ['KYC_PENDING'], ['Your identity check is not complete.'], ['Complete identity verification.']);
   284→    }
   285→    if (arrears > 0n) {
   286→      trace.push({ step: 'hard_stops', result: 'fail', detail: `arrears ${fmtRand(arrears)}` });
   287→      return out('decline', 0n, ['ARREARS_PRESENT'], ['You have an overdue amount on another account.'], [`Clear the overdue ${fmtRand(arrears)} and apply again.`]);
   288→    }
   289→    if (tenure < this.policy.minTenureDays) {
   290→      const more = Math.ceil(this.policy.minTenureDays - tenure);
   291→      trace.push({ step: 'hard_stops', result: 'fail', detail: `tenure ${tenure} days` });
   292→      return out('decline', 0n, ['TENURE_TOO_SHORT'], ['You have not been trading with us long enough yet.'], [`Keep trading for ${more} more days and apply again.`]);
   293→    }
   294→    trace.push({ step: 'hard_stops', result: 'pass', detail: 'kyc, arrears, tenure ok' });
   295→
   296→    // 4. Affordability and limit (illustrative scorecard stand-in: share of recent sales).
   297→    const raw = mulBps(sales, BigInt(this.policy.limitShareOfSalesBps));
   298→    const limit = minOf(raw, this.policy.maxLimit);
   299→    if (limit < this.policy.minLimit) {
   300→      const needed = (this.policy.minLimit * 10000n + BigInt(this.policy.limitShareOfSalesBps) - 1n) / BigInt(this.policy.limitShareOfSalesBps);
   301→      trace.push({ step: 'affordability', result: 'fail', detail: `limit ${fmtRand(limit)} below minimum` });
   302→      return out('decline', 0n, ['AFFORDABILITY_LOW'], ['Your recent sales do not support the smallest advance we offer.'],
   303→        [`Monthly sales of at least ${fmtRand(needed)} would qualify.`]);
   304→    }
   305→    trace.push({ step: 'affordability', result: 'pass', detail: `limit ${fmtRand(limit)}` });
   306→    return out('approve', limit, ['APPROVED'], [`You are approved for up to ${fmtRand(limit)}.`], []);
   307→  }
   308→}

   313→```ts
   314→import { createPrivateKey, createPublicKey, generateKeyPairSync, sign, verify, KeyObject } from 'node:crypto';
   315→
   316→export interface EnvelopePayload {
   317→  envelopeId: string; subject: string; tenant: string; product: string;
   318→  maxOfflineAmountMinor: string; currency: string;
   319→  validFrom: string; expiresAt: string; policyVersion: string; modelVersion: string;
   320→  alg: 'Ed25519'; kid: string;
   321→}
   322→export interface SignedEnvelope { payload: EnvelopePayload; signature: string }
   323→
   324→const canonical = (p: EnvelopePayload) =>
   325→  JSON.stringify(Object.fromEntries(Object.entries(p).sort(([a], [b]) => (a < b ? -1 : 1))));
   326→
   327→export function newKeyPair() {
   328→  const { publicKey, privateKey } = generateKeyPairSync('ed25519');
   329→  return { publicKey, privateKey };
   330→}
   331→export const exportPublic = (k: KeyObject) => k.export({ type: 'spki', format: 'pem' }).toString();
   332→export const importPublic = (pem: string) => createPublicKey(pem);
   333→export const importPrivate = (pem: string) => createPrivateKey(pem);
   334→
   335→export function signEnvelope(payload: EnvelopePayload, privateKey: KeyObject): SignedEnvelope {
   336→  return { payload, signature: sign(null, Buffer.from(canonical(payload)), privateKey).toString('base64') };
   337→}
   338→
   339→export type VerifyResult = { ok: true } | { ok: false; reason: 'UNKNOWN_KEY' | 'BAD_SIGNATURE' | 'NOT_YET_VALID' | 'EXPIRED' | 'WRONG_TENANT' };
   340→
   341→/** What an edge node runs before every offline decision: signature, validity window, tenant, key ID. */
   342→export function verifyEnvelope(env: SignedEnvelope, keys: Map<string, KeyObject>, tenant: string, now = new Date()): VerifyResult {
   343→  const key = keys.get(env.payload.kid);
   344→  if (!key) return { ok: false, reason: 'UNKNOWN_KEY' };
   345→  if (!verify(null, Buffer.from(canonical(env.payload)), key, Buffer.from(env.signature, 'base64'))) return { ok: false, reason: 'BAD_SIGNATURE' };
   346→  if (env.payload.tenant !== tenant) return { ok: false, reason: 'WRONG_TENANT' };
   347→  if (now < new Date(env.payload.validFrom)) return { ok: false, reason: 'NOT_YET_VALID' };
   348→  if (now >= new Date(env.payload.expiresAt)) return { ok: false, reason: 'EXPIRED' };
   349→  return { ok: true };
   350→}
   351→
   352→/** Tracks offline spend against the envelope cap; reported to the core on sync. */
   353→export class EdgeSpendTracker {
   354→  private spent = new Map<string, bigint>();
   355→  tryReserve(env: SignedEnvelope, amount: bigint): boolean {
   356→    const cap = BigInt(env.payload.maxOfflineAmountMinor);
   357→    const next = (this.spent.get(env.payload.envelopeId) ?? 0n) + amount;
   358→    if (amount <= 0n || next > cap) return false;
   359→    this.spent.set(env.payload.envelopeId, next);
   360→    return true;
   361→  }
   362→  spentFor(id: string) { return this.spent.get(id) ?? 0n; }
   363→}

   376→
   377→export interface GoPayConfig {

   384→

   389→export const acct = {

   397→export interface SettleResult { fee: bigint; net: bigint; repaid: bigint; payout: bigint; replayed: boolean }
   398→
   399→export async function settleSale(ledger: LedgerService, cfg: GoPayConfig, s: Sale): Promise<SettleResult> {
   400→  if (cfg.route === 'split' && !cfg.goPayLive) throw new GoPayNotLiveError();
   401→  if (s.amount <= 0n) throw new RangeError('sale amount must be positive');
   402→  const cur = s.currency ?? 'ZAR';
   403→  const fee = mulBps(s.amount, BigInt(MDR_BPS[s.rail]));
   404→  const net = s.amount - fee;
   405→  const base = { tenantId: s.tenantId, correlationId: s.saleId, currency: cur };
   406→  const key = (step: string) => `${s.saleId}:${step}`;
   407→
   408→  // Replay safety: if the final step exists, rebuild the result from stored entries and post nothing.
   409→  const done = await ledger.get(s.tenantId, key('payout'));
   410→  if (done) {
   411→    const rep = await ledger.get(s.tenantId, key('repayment'));
   412→    const repaid = rep ? rep.lines.find((l) => l.account === acct.wallet(s.merchantId))!.amount : 0n;
   413→    return { fee, net, repaid, payout: net - repaid, replayed: true };
   414→  }
   415→
   416→  await ledger.post({ ...base, idempotencyKey: key('captured'), reason: 'payment_captured',
   417→    lines: [{ account: acct.clearing, amount: s.amount }, { account: acct.wallet(s.merchantId), amount: -s.amount }] });
   418→  await ledger.post({ ...base, idempotencyKey: key('mdr'), reason: 'merchant_fee',
   419→    lines: [{ account: acct.wallet(s.merchantId), amount: fee }, { account: acct.mdr, amount: -fee }] });
   420→  await ledger.post({ ...base, idempotencyKey: key('settled'), reason: 'partner_settlement',
   421→    lines: [{ account: acct.trust, amount: s.amount }, { account: acct.clearing, amount: -s.amount }] });
   422→
   423→  let repaid = 0n;
   424→  if (cfg.route === 'split') {
   425→    const outstanding = await ledger.balance(s.tenantId, acct.loan(s.merchantId));
   426→    repaid = splitAtSettlement(net, cfg.shareBps, outstanding);
   427→    if (repaid > 0n) {
   428→      await ledger.post({ ...base, idempotencyKey: key('repayment'), reason: 'repayment_split',
   429→        lines: [{ account: acct.wallet(s.merchantId), amount: repaid }, { account: acct.loan(s.merchantId), amount: -repaid }] });
   430→    }
   431→  }
   432→  const payout = net - repaid;
   433→  await ledger.post({ ...base, idempotencyKey: key('payout'), reason: 'merchant_payout',
   434→    lines: [{ account: acct.wallet(s.merchantId), amount: payout }, { account: acct.trust, amount: -payout }] });
   435→  return { fee, net, repaid, payout, replayed: false };
   436→}
   437→
   438→/** Wave 1 route: scheduled debit order collected through a licensed partner. */
   439→export async function collectDebit(ledger: LedgerService, p: { tenantId: string; merchantId: string; debitId: string; instalment: bigint }) {
   440→  const outstanding = await ledger.balance(p.tenantId, acct.loan(p.merchantId));
   441→  const amount = minOf(outstanding, p.instalment);
   442→  if (amount <= 0n) return { collected: 0n };
   443→  await ledger.post({ tenantId: p.tenantId, idempotencyKey: `${p.debitId}:collected`, reason: 'debit_order_collected',
   444→    correlationId: p.debitId, currency: 'ZAR',
   445→    lines: [{ account: acct.trust, amount }, { account: acct.loan(p.merchantId), amount: -amount }] });
   446→  return { collected: amount };
   447→}
   448→
   449→/** A failed debit order posts nothing to the ledger: it raises arrears and starts the collections workflow. */
   450→export async function failedDebit(ledger: LedgerService, p: { tenantId: string; merchantId: string; instalment: bigint }) {
   451→  const outstanding = await ledger.balance(p.tenantId, acct.loan(p.merchantId));
   452→  return { arrears: minOf(outstanding, p.instalment), nextAction: 'collections_contact_then_hardship_offer' as const };
   453→}

   459→import { uuidv7 } from '../shared/ids';
   460→
   461→/** Signed amounts: debit positive, credit negative. Every entry must sum to zero. */
   462→export interface Line { account: string; amount: bigint }
   463→export interface EntryInput {
   464→  tenantId: string;
   465→  idempotencyKey: string;
   466→  reason: string;
   467→  correlationId: string;
   468→  currency: string;
   469→  lines: Line[];
   470→}
   471→export interface Entry extends EntryInput { id: string; postedAt: Date }
   472→export interface OutboxEvent { id: string; topic: string; payload: unknown }
   473→
   474→/** Port. A Postgres/Prisma adapter must write the entry and outbox row in ONE transaction. */
   475→export interface LedgerRepo {
   476→  findByKey(tenantId: string, key: string): Promise<Entry | undefined>;
   477→  appendWithOutbox(entry: Entry, event: OutboxEvent): Promise<void>;
   478→  entries(tenantId: string): Promise<readonly Entry[]>;
   479→}
   480→
   481→export class LedgerError extends Error {
   482→  constructor(public readonly code: string, message: string) { super(message); }
   483→}
   484→
   485→const fingerprint = (i: EntryInput) =>
   486→  [i.reason, i.currency, ...i.lines.map((l) => `${l.account}:${l.amount}`)].join('|');
   487→
   488→export class InMemoryLedgerRepo implements LedgerRepo {
   489→  private byKey = new Map<string, Entry>();
   490→  private byTenant = new Map<string, Entry[]>();
   491→  readonly outbox: OutboxEvent[] = [];
   492→
   493→  async findByKey(t: string, k: string) { return this.byKey.get(`${t}/${k}`); }
   494→  async appendWithOutbox(e: Entry, ev: OutboxEvent) {
   495→    this.byKey.set(`${e.tenantId}/${e.idempotencyKey}`, e);
   496→    const list = this.byTenant.get(e.tenantId) ?? [];
   497→    list.push(e);
   498→    this.byTenant.set(e.tenantId, list);
   499→    this.outbox.push(ev);
   500→  }
   501→  async entries(t: string) { return this.byTenant.get(t) ?? []; }
   502→}
   503→
   504→export class LedgerService {
   505→  constructor(private readonly repo: LedgerRepo) {}
   506→
   507→  async post(input: EntryInput): Promise<{ entry: Entry; replayed: boolean }> {
   508→    if (!/^[A-Z]{3}$/.test(input.currency)) throw new LedgerError('BAD_CURRENCY', 'currency must be ISO 4217');
   509→    if (input.lines.length < 2) throw new LedgerError('TOO_FEW_LINES', 'an entry needs at least two lines');
   510→    let sum = 0n;
   511→    for (const l of input.lines) {
   512→      if (!l.account) throw new LedgerError('NO_ACCOUNT', 'every line needs an account');
   513→      if (l.amount === 0n) throw new LedgerError('ZERO_AMOUNT', 'zero-amount lines are not allowed');
   514→      sum += l.amount;
   515→    }
   516→    if (sum !== 0n) throw new LedgerError('UNBALANCED', `entry sums to ${sum}, expected 0`);
   517→
   518→    const existing = await this.repo.findByKey(input.tenantId, input.idempotencyKey);
   519→    if (existing) {
   520→      if (fingerprint(existing) !== fingerprint(input)) {
   521→        throw new LedgerError('IDEMPOTENCY_CONFLICT', 'same key reused with different content');
   522→      }
   523→      return { entry: existing, replayed: true };
   524→    }
   525→    const entry: Entry = { ...input, id: uuidv7(), postedAt: new Date() };
   526→    await this.repo.appendWithOutbox(entry, {
   527→      id: uuidv7(),
   528→      topic: 'ledger.entry.posted',
   529→      payload: { entryId: entry.id, tenantId: entry.tenantId, reason: entry.reason, correlationId: entry.correlationId },
   530→    });
   531→    return { entry, replayed: false };
   532→  }
   533→
   534→  async get(tenantId: string, key: string) { return this.repo.findByKey(tenantId, key); }
   535→
   536→  async balance(tenantId: string, account: string): Promise<bigint> {
   537→    let b = 0n;
   538→    for (const e of await this.repo.entries(tenantId)) for (const l of e.lines) if (l.account === account) b += l.amount;
   539→    return b;
   540→  }
   541→
   542→  /** Sum of all lines across the tenant. Must always be zero. */
   543→  async trialBalanceTotal(tenantId: string): Promise<bigint> {
   544→    let t = 0n;
   545→    for (const e of await this.repo.entries(tenantId)) for (const l of e.lines) t += l.amount;
   546→    return t;
   547→  }
   548→}

   554→import type { Entry, LedgerRepo, OutboxEvent } from './ledger';
   555→
   556→/** Minimal structural types so this adapter compiles without importing the generated Prisma client. */
   557→/* eslint-disable @typescript-eslint/no-explicit-any */
   558→export interface TshepoTx {
   559→  tshepoJournalEntry: { findUnique(a: any): Promise<any>; create(a: any): Promise<any>; findMany(a: any): Promise<any[]> };
   560→  tshepoAccount: { upsert(a: any): Promise<{ id: string }> };
   561→  tshepoOutbox: { create(a: any): Promise<any> };
   562→}
   563→export interface TshepoDb extends TshepoTx { $transaction<T>(fn: (tx: TshepoTx) => Promise<T>): Promise<T> }
   564→
   565→const toEntry = (r: any): Entry => ({
   566→  id: r.id, tenantId: r.tenantId, idempotencyKey: r.idempotencyKey, reason: r.reason,
   567→  correlationId: r.correlationId, currency: r.currency, postedAt: r.postedAt,
   568→  lines: r.lines.map((l: any) => ({ account: l.account.code, amount: BigInt(l.amountMinor) })),
   569→});
   570→const include = { lines: { include: { account: true } } };
   571→
   572→/** Prisma-backed ledger. The entry, its lines and its outbox row commit in ONE transaction. */
   573→export class PrismaLedgerRepo implements LedgerRepo {
   574→  constructor(private readonly db: TshepoDb) {}
   575→
   576→  async findByKey(tenantId: string, key: string) {
   577→    const r = await this.db.tshepoJournalEntry.findUnique({ where: { tenantId_idempotencyKey: { tenantId, idempotencyKey: key } }, include });
   578→    return r ? toEntry(r) : undefined;
   579→  }
   580→
   581→  async appendWithOutbox(entry: Entry, event: OutboxEvent) {
   582→    await this.db.$transaction(async (tx) => {
   583→      const lines = [];
   584→      for (const l of entry.lines) {
   585→        const acc = await tx.tshepoAccount.upsert({
   586→          where: { tenantId_code: { tenantId: entry.tenantId, code: l.account } },
   587→          create: { id: `${entry.tenantId}:${l.account}`, tenantId: entry.tenantId, code: l.account, kind: l.account.split(':')[0], currency: entry.currency },
   588→          update: {},
   589→        });
   590→        lines.push({ id: `${entry.id}:${lines.length}`, accountId: acc.id, amountMinor: l.amount, currency: entry.currency });
   591→      }
   592→      await tx.tshepoJournalEntry.create({
   593→        data: { id: entry.id, tenantId: entry.tenantId, idempotencyKey: entry.idempotencyKey, reason: entry.reason,
   594→          correlationId: entry.correlationId, currency: entry.currency, postedAt: entry.postedAt, lines: { create: lines } },
   595→      });
   596→      await tx.tshepoOutbox.create({ data: { id: event.id, tenantId: entry.tenantId, topic: event.topic, payload: Buffer.from(JSON.stringify(event.payload)) } });
   597→    });
   598→  }
   599→
   600→  async entries(tenantId: string) {
   601→    const rows = await this.db.tshepoJournalEntry.findMany({ where: { tenantId }, include, orderBy: { postedAt: 'asc' } });
   602→    return rows.map(toEntry);
   603→  }
   604→}

   609→```ts
   610→import { mulBps, minOf } from '../shared/money';
   611→
   612→export interface Bucket { name: string; due: bigint }
   613→
   614→/** Allocate a payment across buckets in the configured order. Order is product config approved by compliance. */
   615→export function allocateWaterfall(amount: bigint, buckets: Bucket[]) {
   616→  if (amount < 0n) throw new RangeError('amount must be non-negative');
   617→  let left = amount;
   618→  const allocations: Record<string, bigint> = {};
   619→  for (const b of buckets) {
   620→    const take = minOf(left, b.due < 0n ? 0n : b.due);
   621→    allocations[b.name] = take;
   622→    left -= take;
   623→  }
   624→  return { allocations, unapplied: left };
   625→}
   626→
   627→/** GoPay settlement split: share of net sale, never more than the amount outstanding. */
   628→export function splitAtSettlement(net: bigint, shareBps: number, outstanding: bigint): bigint {
   629→  if (shareBps < 0 || shareBps > 10000) throw new RangeError('shareBps out of range');
   630→  if (outstanding <= 0n || net <= 0n) return 0n;
   631→  return minOf(outstanding, mulBps(net, BigInt(shareBps)));
   632→}

   638→import { randomBytes } from 'node:crypto';
   639→
   640→/** UUIDv7: 48-bit millisecond timestamp, then random bits. Sortable by creation time. */
   641→export function uuidv7(now: number = Date.now()): string {
   642→  const b = randomBytes(16);
   643→  const ts = BigInt(now);
   644→  for (let i = 0; i < 6; i++) b[i] = Number((ts >> BigInt(40 - 8 * i)) & 0xffn);
   645→  b[6] = (b[6]! & 0x0f) | 0x70;
   646→  b[8] = (b[8]! & 0x3f) | 0x80;
   647→  const h = b.toString('hex');
   648→  return `${h.slice(0, 8)}-${h.slice(8, 12)}-${h.slice(12, 16)}-${h.slice(16, 20)}-${h.slice(20)}`;
   649→}

   655→/** All money is integer minor units (cents) held as bigint. Never floating point. */
   656→export type Minor = bigint;
   657→
   658→/** amount * bps / 10_000, rounded half-up. Amounts must be non-negative. */
   659→export function mulBps(amount: Minor, bps: bigint): Minor {
   660→  if (amount < 0n || bps < 0n) throw new RangeError('mulBps expects non-negative inputs');
   661→  return (amount * bps + 5000n) / 10000n;
   662→}
   663→
   664→export const minOf = (a: Minor, b: Minor): Minor => (a < b ? a : b);
   665→
   666→export function fmtRand(c: Minor): string {
   667→  const neg = c < 0n;
   668→  const abs = neg ? -c : c;
   669→  const whole = (abs / 100n).toString().replace(/\B(?=(\d{3})+(?!\d))/g, ',');
   670→  return `${neg ? '-' : ''}R${whole}.${(abs % 100n).toString().padStart(2, '0')}`;
   671→}

   688→export interface AdvanceRequest {
   689→  tenantId: string; subjectId: string; productId: string; amountMinor: bigint; requestId: string; history: History; now?: Date;
   690→  /** Where to deliver: msisdn for airtime and data, meterNumber for electricity. */

   693→
   694→/** Issue the advance, deliver the airtime or token, and keep the books right whatever the delivery outcome. */
   695→export class AdvanceDelivery {
   696→  constructor(private credit: TopUpCredit, private router: VendingRouter) {}
   697→
   698→  async request(a: AdvanceRequest): Promise<DeliveryResult> {
   699→    const p = this.credit.getProduct(a.productId);
   700→    let issued;
   701→    try { issued = await this.credit.issue(a); }
   702→    catch (e: any) { return { status: 'declined', code: e?.code ?? 'ERROR' }; }
   703→
   704→    const out = await this.router.vend({
   705→      reference: a.requestId, country: p.jurisdiction, commodity: p.commodity, amountMinor: a.amountMinor, currency: p.currency, ...a.target,
   706→    });
   707→    if (out.status === 'success') return { status: 'delivered', token: out.token, providerRef: out.providerRef, replayed: issued.replayed };

   709→    await this.credit.markPending(a.tenantId, a.requestId);
   710→    return { status: 'pending', reason: out.reason };
   711→  }

   714→  async settlePending(tenantId: string, requestId: string): Promise<DeliveryResult> {

   716→    if (out.status === 'success') { await this.credit.confirm(tenantId, requestId); return { status: 'delivered', token: out.token, providerRef: out.providerRef, replayed: false }; }
   717→    if (out.status === 'failed') { await this.credit.cancel(tenantId, requestId); return { status: 'cancelled', code: out.code }; }

   769→### `src/lib/tshepo/topup/products.ts`
   770→
   771→```ts
   772→import { mulBps } from '../shared/money';
   773→
   774→export type Jurisdiction = 'ZA' | 'MW';
   775→export type Commodity = 'airtime' | 'data' | 'electricity';
   776→
   777→/** How the access fee is charged. Illustrative defaults are in presets.ts and must be confirmed with each partner. */
   778→export type FeeModel =
   779→  | { kind: 'flat'; feeMinor: bigint }                                   // e.g. a fixed access fee per advance
   780→  | { kind: 'per_block'; blockMinor: bigint; feePerBlockMinor: bigint }  // e.g. fee per R10 or part thereof
   781→  | { kind: 'percent_bps'; bps: number };                                // e.g. 10% = 1000 bps
   782→
   783→/** full_first: the whole top-up goes to the advance until it is repaid. capped_share: only a share of each purchase. */
   784→export interface RecoveryRule { mode: 'full_first' | 'capped_share'; capBps?: number }
   785→
   786→export interface Product {
   787→  id: string;
   788→  jurisdiction: Jurisdiction;
   789→  commodity: Commodity;
   790→  currency: string;
   791→  /** The advance amounts the customer may choose from (the USSD or app menu). */
   792→  menuMinor: bigint[];
   793→  fee: FeeModel;
   794→  /** Hard ceiling on the fee for one advance, set from the rules that apply to the product. Checked at quote time. */
   795→  maxFeeMinor?: bigint;
   796→  recovery: RecoveryRule;
   797→  minTenureDays: number;
   798→  /** Eligibility: at least minMonths of the last 3 calendar months with recharges totalling minMonthlyRechargeMinor. */
   799→  minMonths: number;
   800→  minMonthlyRechargeMinor: bigint;
   801→  /** Largest advance as a share of average monthly recharge, in basis points. */
   802→  limitShareBps: number;
   803→  /** After this many days unrecovered, the overdue action applies. */
   804→  dueAfterDays: number;
   805→  overdueAction: 'add_to_bill' | 'write_off';
   806→}
   807→
   808→export function quoteFee(p: Product, principal: bigint): bigint {
   809→  if (principal <= 0n) throw new RangeError('principal must be positive');
   810→  let fee: bigint;
   811→  switch (p.fee.kind) {
   812→    case 'flat': fee = p.fee.feeMinor; break;
   813→    case 'per_block': fee = ((principal + p.fee.blockMinor - 1n) / p.fee.blockMinor) * p.fee.feePerBlockMinor; break;
   814→    case 'percent_bps': fee = mulBps(principal, BigInt(p.fee.bps)); break;
   815→  }
   816→  if (p.maxFeeMinor !== undefined && fee > p.maxFeeMinor) {
   817→    throw new RangeError(`fee ${fee} exceeds the product ceiling ${p.maxFeeMinor}`);
   818→  }
   819→  return fee;
   820→}
   821→
   822→export interface History {
   823→  tenureDays: number;
   824→  recharges: { amountMinor: bigint; at: Date }[];
   825→  arrearsMinor: bigint;
   826→  /** Electricity: meter registered and linked to the customer. Airtime: SIM registered. */
   827→  identityOk: boolean;
   828→}
   829→
   830→export interface Eligibility { ok: boolean; maxMinor: bigint; reasons: string[] }
   831→
   832→export function assessEligibility(p: Product, h: History, now: Date): Eligibility {
   833→  const reasons: string[] = [];
   834→  if (!h.identityOk) reasons.push('IDENTITY_NOT_VERIFIED');
   835→  if (h.arrearsMinor > 0n) reasons.push('ARREARS_PRESENT');
   836→  if (h.tenureDays < p.minTenureDays) reasons.push('TENURE_TOO_SHORT');
   837→
   838→  const since = new Date(now.getTime() - 90 * 86400000);
   839→  const byMonth = new Map<string, bigint>();
   840→  let total = 0n;
   841→  for (const r of h.recharges) {
   842→    if (r.at < since || r.at > now) continue;
   843→    const k = `${r.at.getUTCFullYear()}-${r.at.getUTCMonth()}`;
   844→    byMonth.set(k, (byMonth.get(k) ?? 0n) + r.amountMinor);
   845→    total += r.amountMinor;
   846→  }
   847→  const goodMonths = [...byMonth.values()].filter((v) => v >= p.minMonthlyRechargeMinor).length;
   848→  if (goodMonths < p.minMonths) reasons.push('RECHARGE_HISTORY_TOO_THIN');
   849→
   850→  const avgMonthly = total / 3n;
   851→  const cap = mulBps(avgMonthly, BigInt(p.limitShareBps));
   852→  const eligibleMenu = p.menuMinor.filter((m) => m <= cap);
   853→  const maxMinor = eligibleMenu.length ? eligibleMenu.reduce((a, b) => (a > b ? a : b)) : 0n;
   854→  if (maxMinor === 0n && !reasons.includes('RECHARGE_HISTORY_TOO_THIN')) reasons.push('LIMIT_BELOW_SMALLEST_ADVANCE');
   855→  return { ok: reasons.length === 0, maxMinor, reasons };
   856→}

   862→import { LedgerService } from '../ledger/ledger';

   864→import { mulBps, minOf } from '../shared/money';

   867→export interface AdvanceRecord {
   868→  id: string; tenantId: string; subjectId: string; productId: string; requestId: string;
   869→  principal: bigint; fee: bigint; recovered: bigint; issuedAt: Date;
   870→  status: 'pending' | 'open' | 'repaid' | 'billed' | 'written_off' | 'cancelled';
   871→}
   872→export interface AdvanceRepo {
   873→  byRequest(tenantId: string, requestId: string): Promise<AdvanceRecord | undefined>;
   874→  openFor(tenantId: string, subjectId: string): Promise<AdvanceRecord | undefined>;
   875→  save(r: AdvanceRecord): Promise<void>;
   876→}
   877→export class InMemoryAdvanceRepo implements AdvanceRepo {
   878→  private rows: AdvanceRecord[] = [];
   879→  async byRequest(t: string, id: string) { return this.rows.find((r) => r.tenantId === t && r.requestId === id); }
   880→  async openFor(t: string, s: string) { return this.rows.find((r) => r.tenantId === t && r.subjectId === s && (r.status === 'open' || r.status === 'pending')); }
   881→  async save(r: AdvanceRecord) { const i = this.rows.findIndex((x) => x.id === r.id); if (i >= 0) this.rows[i] = r; else this.rows.push(r); }
   882→}
   883→
   884→export class TopUpCreditError extends Error {
   885→  constructor(public readonly code: string, message: string) { super(message); }
   886→}
   887→
   888→export const topupAcct = {
   889→  receivable: (subject: string) => `advance_receivable:${subject}`,
   890→  supplier: (commodity: string) => `supplier_payable:${commodity}`,
   891→  fee: 'fee_income:topup',
   892→  clearing: 'collections_clearing',
   893→  badDebt: 'bad_debt_expense',
   894→};
   895→
   896→export class TopUpCredit {
   897→  private products: Map<string, Product>;
   898→  constructor(private ledger: LedgerService, private repo: AdvanceRepo, products: Product[]) {
   899→    this.products = new Map(products.map((p) => [p.id, p]));
   900→  }
   901→  getProduct(id: string) { return this.product(id); }
   902→  private product(id: string) {
   903→    const p = this.products.get(id);
   904→    if (!p) throw new TopUpCreditError('UNKNOWN_PRODUCT', id);
   905→    return p;
   906→  }
   907→
   908→  /** Everything the customer sees before accepting: amount, fee, total to repay. */
   909→  quote(productId: string, amountMinor: bigint) {
   910→    const p = this.product(productId);
   911→    if (!p.menuMinor.includes(amountMinor)) throw new TopUpCreditError('NOT_ON_MENU', 'amount is not an offered advance');
   912→    const fee = quoteFee(p, amountMinor);
   913→    return { principal: amountMinor, fee, totalDue: amountMinor + fee };
   914→  }
   915→
   916→  /** Issue an advance. The commodity (airtime, data or a meter token) is delivered by the operator or vendor adapter. */
   917→  async issue(a: { tenantId: string; subjectId: string; productId: string; amountMinor: bigint; requestId: string; history: History; now?: Date }) {
   918→    const now = a.now ?? new Date();
   919→    const existing = await this.repo.byRequest(a.tenantId, a.requestId);
   920→    if (existing) return { record: existing, replayed: true };
   921→    const p = this.product(a.productId);
   922→    if (await this.repo.openFor(a.tenantId, a.subjectId)) throw new TopUpCreditError('OPEN_ADVANCE', 'one advance at a time');
   923→    const el = assessEligibility(p, a.history, now);
   924→    if (!el.ok) throw new TopUpCreditError('NOT_ELIGIBLE', el.reasons.join(','));
   925→    if (a.amountMinor > el.maxMinor) throw new TopUpCreditError('ABOVE_LIMIT', `limit is ${el.maxMinor}`);
   926→    const q = this.quote(a.productId, a.amountMinor);
   927→    await this.ledger.post({
   928→      tenantId: a.tenantId, idempotencyKey: `${a.requestId}:issued`, reason: 'advance_issued', correlationId: a.requestId, currency: p.currency,
   929→      lines: [
   930→        { account: topupAcct.receivable(a.subjectId), amount: q.totalDue },
   931→        { account: topupAcct.supplier(p.commodity), amount: -q.principal },
   932→        { account: topupAcct.fee, amount: -q.fee },
   933→      ],
   934→    });
   935→    const record: AdvanceRecord = { id: uuidv7(), tenantId: a.tenantId, subjectId: a.subjectId, productId: p.id, requestId: a.requestId,
   936→      principal: q.principal, fee: q.fee, recovered: 0n, issuedAt: now, status: 'open' };
   937→    await this.repo.save(record);
   938→    return { record, replayed: false };
   939→  }
   940→
   941→  /**
   942→   * Called on every top-up or purchase by the subject (the clawback). Takes what the product rule allows,
   943→   * posts it, and tells the vending adapter how much of the purchase to pass through as normal value.
   944→   */
   945→  async recover(a: { tenantId: string; subjectId: string; topupRef: string; topupAmountMinor: bigint }) {
   946→    if (a.topupAmountMinor <= 0n) throw new RangeError('top-up amount must be positive');
   947→    const key = `${a.topupRef}:recovered`;
   948→    const done = await this.ledger.get(a.tenantId, key);
   949→    if (done) {
   950→      const taken = -done.lines.find((l) => l.account === topupAcct.receivable(a.subjectId))!.amount;
   951→      return { taken, passThrough: a.topupAmountMinor - taken, replayed: true, repaid: false };
   952→    }
   953→    const open = await this.repo.openFor(a.tenantId, a.subjectId);
   954→    if (!open || open.status !== 'open') return { taken: 0n, passThrough: a.topupAmountMinor, replayed: false, repaid: false };
   955→    const p = this.product(open.productId);
   956→    const due = open.principal + open.fee - open.recovered;
   957→    const room = p.recovery.mode === 'capped_share' ? mulBps(a.topupAmountMinor, BigInt(p.recovery.capBps ?? 10000)) : a.topupAmountMinor;
   958→    const taken = minOf(due, room);
   959→    if (taken > 0n) {
   960→      await this.ledger.post({
   961→        tenantId: a.tenantId, idempotencyKey: key, reason: 'advance_recovered', correlationId: a.topupRef, currency: p.currency,
   962→        lines: [{ account: topupAcct.clearing, amount: taken }, { account: topupAcct.receivable(a.subjectId), amount: -taken }],
   963→      });
   964→      open.recovered += taken;
   965→      if (open.recovered === open.principal + open.fee) open.status = 'repaid';
   966→      await this.repo.save(open);
   967→    }
   968→    return { taken, passThrough: a.topupAmountMinor - taken, replayed: false, repaid: open.status === 'repaid' };
   969→  }
   970→
   971→  /** Delivery outcome unknown: keep the advance, block a second one, and do not recover until delivery is confirmed. */
   972→  async markPending(tenantId: string, requestId: string) {
   973→    const r = await this.repo.byRequest(tenantId, requestId);
   974→    if (r && r.status === 'open') { r.status = 'pending'; await this.repo.save(r); }
   975→    return r;
   976→  }
   977→  /** Delivery confirmed after an unknown outcome. */
   978→  async confirm(tenantId: string, requestId: string) {
   979→    const r = await this.repo.byRequest(tenantId, requestId);
   980→    if (r && r.status === 'pending') { r.status = 'open'; await this.repo.save(r); }
   981→    return r;
   982→  }
   983→  /** Delivery definitely failed: reverse the issue entry exactly, so nothing is owed. Not allowed once any recovery has happened. */
   984→  async cancel(tenantId: string, requestId: string) {
   985→    const r = await this.repo.byRequest(tenantId, requestId);
   986→    if (!r || r.status === 'cancelled') return r;
   987→    if (r.recovered > 0n || (r.status !== 'open' && r.status !== 'pending')) throw new TopUpCreditError('CANNOT_CANCEL', 'advance already in repayment or closed');
   988→    const p = this.product(r.productId);
   989→    await this.ledger.post({
   990→      tenantId, idempotencyKey: `${requestId}:cancelled`, reason: 'advance_cancelled', correlationId: requestId, currency: p.currency,
   991→      lines: [
   992→        { account: topupAcct.supplier(p.commodity), amount: r.principal },
   993→        { account: topupAcct.fee, amount: r.fee },
   994→        { account: topupAcct.receivable(r.subjectId), amount: -(r.principal + r.fee) },
   995→      ],
   996→    });
   997→    r.status = 'cancelled'; await this.repo.save(r);
   998→    return r;
   999→  }
  1000→
  1001→  /** Daily job: what to do with advances not recovered within the product's period. */
  1002→  async applyOverdue(tenantId: string, subjectId: string, now: Date) {
  1003→    const open = await this.repo.openFor(tenantId, subjectId);
  1004→    if (!open || open.status !== 'open') return { action: 'none' as const };
  1005→    const p = this.product(open.productId);
  1006→    if (now.getTime() - open.issuedAt.getTime() < p.dueAfterDays * 86400000) return { action: 'none' as const };
  1007→    const owing = open.principal + open.fee - open.recovered;
  1008→    if (p.overdueAction === 'add_to_bill') {
  1009→      open.status = 'billed'; await this.repo.save(open);
  1010→      return { action: 'add_to_bill' as const, amountMinor: owing };
  1011→    }
  1012→    await this.ledger.post({
  1013→      tenantId, idempotencyKey: `${open.requestId}:written_off`, reason: 'advance_written_off', correlationId: open.requestId, currency: p.currency,
  1014→      lines: [{ account: topupAcct.badDebt, amount: owing }, { account: topupAcct.receivable(subjectId), amount: -owing }],
  1015→    });
  1016→    open.status = 'written_off'; await this.repo.save(open);
  1017→    return { action: 'write_off' as const, amountMinor: owing };
  1018→  }
  1019→}

  1063→import { Country, MeterInfo, VendCommodity, VendOutcome, VendRequest, VendingProvider } from './types';
  1064→
  1065→export interface HttpCall { method: 'GET' | 'POST'; path: string; body?: unknown }
  1066→
  1067→/**
  1068→ * A provider integration is configuration: how to build each call and how to read each reply.
  1069→ * Onboarding a new aggregator means writing one config object once its partner documentation is in hand.
  1070→ */
  1071→export interface HttpAdapterConfig {
  1072→  id: string;
  1073→  baseUrl: string;
  1074→  headers: () => Record<string, string>;
  1075→  supported: { commodity: VendCommodity; country: Country }[];
  1076→  buildVend(req: VendRequest): HttpCall;
  1077→  mapVend(httpStatus: number, body: any, req: VendRequest): VendOutcome;
  1078→  buildStatus?(reference: string): HttpCall;
  1079→  mapStatus?(httpStatus: number, body: any, reference: string): VendOutcome;
  1080→  buildValidate?(meterNumber: string, country: Country, municipality?: string): HttpCall;
  1081→  mapValidate?(httpStatus: number, body: any): MeterInfo;
  1082→  timeoutMs?: number;
  1083→  fetchImpl?: typeof fetch;
  1084→}
  1085→
  1086→export class HttpVendingAdapter implements VendingProvider {
  1087→  readonly id: string;
  1088→  constructor(private cfg: HttpAdapterConfig) { this.id = cfg.id; }
  1089→
  1090→  supports(c: VendCommodity, country: Country) { return this.cfg.supported.some((s) => s.commodity === c && s.country === country); }
  1091→
  1092→  private async call(c: HttpCall): Promise<{ status: number; body: any }> {
  1093→    const f = this.cfg.fetchImpl ?? fetch;
  1094→    const ctl = new AbortController();
  1095→    const timer = setTimeout(() => ctl.abort(), this.cfg.timeoutMs ?? 15_000);
  1096→    try {
  1097→      const res = await f(this.cfg.baseUrl + c.path, {
  1098→        method: c.method, signal: ctl.signal,
  1099→        headers: { 'content-type': 'application/json', accept: 'application/json', ...this.cfg.headers() },
  1100→        body: c.body === undefined ? undefined : JSON.stringify(c.body),
  1101→      });
  1102→      let body: any = undefined;
  1103→      try { body = await res.json(); } catch { body = undefined; }
  1104→      return { status: res.status, body };
  1105→    } finally { clearTimeout(timer); }
  1106→  }
  1107→
  1108→  /** No answer, a timeout or a server error means we cannot know whether it was delivered, so the result is unknown. */
  1109→  async vend(req: VendRequest): Promise<VendOutcome> {
  1110→    let r: { status: number; body: any };
  1111→    try { r = await this.call(this.cfg.buildVend(req)); }
  1112→    catch (e) { return { status: 'unknown', providerId: this.id, reason: `NO_RESPONSE: ${String(e)}` }; }
  1113→    if (r.status >= 500) return { status: 'unknown', providerId: this.id, reason: `HTTP_${r.status}` };
  1114→    if (r.status === 429) return { status: 'failed', providerId: this.id, code: 'RATE_LIMITED', message: 'rate limited', retryable: true };
  1115→    return this.cfg.mapVend(r.status, r.body, req);
  1116→  }
  1117→
  1118→  async status(reference: string): Promise<VendOutcome> {
  1119→    if (!this.cfg.buildStatus || !this.cfg.mapStatus) return { status: 'unknown', providerId: this.id, reason: 'STATUS_ENDPOINT_NOT_CONFIGURED' };
  1120→    try { const r = await this.call(this.cfg.buildStatus(reference)); return this.cfg.mapStatus(r.status, r.body, reference); }
  1121→    catch (e) { return { status: 'unknown', providerId: this.id, reason: `NO_RESPONSE: ${String(e)}` }; }
  1122→  }
  1123→
  1124→  async validateMeter(meterNumber: string, country: Country, municipality?: string): Promise<MeterInfo> {
  1125→    if (!this.cfg.buildValidate || !this.cfg.mapValidate) return { valid: false };
  1126→    const r = await this.call(this.cfg.buildValidate(meterNumber, country, municipality));
  1127→    return this.cfg.mapValidate(r.status, r.body);
  1128→  }
  1129→}

  1227→
  1228→```ts
  1229→import { VendOutcome, VendRequest, VendingProvider } from './types';
  1230→
  1231→export interface VendStore {
  1232→  get(reference: string): Promise<{ outcome: VendOutcome } | undefined>;
  1233→  put(reference: string, outcome: VendOutcome): Promise<void>;
  1234→}
  1235→export class InMemoryVendStore implements VendStore {
  1236→  private m = new Map<string, { outcome: VendOutcome }>();
  1237→  async get(r: string) { return this.m.get(r); }
  1238→  async put(r: string, outcome: VendOutcome) { this.m.set(r, { outcome }); }
  1239→}
  1240→
  1241→export interface RoutedProvider { provider: VendingProvider; priority: number }
  1242→export interface RouterOptions { failureThreshold?: number; cooldownMs?: number; now?: () => number }
  1243→
  1244→/**
  1245→ * Routes a delivery to the best available provider and fails over ONLY when the first provider definitely
  1246→ * did not deliver. An unknown outcome (timeout, unreadable reply) stops the chain: the item might have been
  1247→ * delivered, so it is looked up, never sent to a second provider.
  1248→ */
  1249→export class VendingRouter {
  1250→  private health = new Map<string, { failures: number; openUntil: number }>();
  1251→  private threshold: number; private cooldown: number; private now: () => number;
  1252→  constructor(private providers: RoutedProvider[], private store: VendStore, opts: RouterOptions = {}) {
  1253→    this.threshold = opts.failureThreshold ?? 3; this.cooldown = opts.cooldownMs ?? 60_000; this.now = opts.now ?? Date.now;
  1254→  }
  1255→
  1256→  private circuitOpen(id: string) { const h = this.health.get(id); return !!h && h.openUntil > this.now(); }
  1257→  private record(id: string, ok: boolean) {
  1258→    const h = this.health.get(id) ?? { failures: 0, openUntil: 0 };
  1259→    if (ok) { h.failures = 0; h.openUntil = 0; } else if (++h.failures >= this.threshold) h.openUntil = this.now() + this.cooldown;
  1260→    this.health.set(id, h);
  1261→  }
  1262→
  1263→  async vend(req: VendRequest): Promise<VendOutcome> {
  1264→    const prior = await this.store.get(req.reference);
  1265→    if (prior) return prior.outcome.status === 'unknown' ? this.resolve(req.reference) : prior.outcome;
  1266→
  1267→    const candidates = this.providers
  1268→      .filter((p) => p.provider.supports(req.commodity, req.country) && !this.circuitOpen(p.provider.id))
  1269→      .sort((a, b) => a.priority - b.priority);
  1270→    let last: VendOutcome = { status: 'failed', providerId: 'router', code: 'NO_PROVIDER', message: 'no available provider', retryable: false };
  1271→
  1272→    for (const { provider } of candidates) {
  1273→      let out: VendOutcome;
  1274→      try { out = await provider.vend(req); }
  1275→      catch (e) { out = { status: 'unknown', providerId: provider.id, reason: String(e) }; }
  1276→      if (out.status === 'success') { this.record(provider.id, true); await this.store.put(req.reference, out); return out; }
  1277→      if (out.status === 'unknown') { this.record(provider.id, false); await this.store.put(req.reference, out); return out; }
  1278→      this.record(provider.id, false);
  1279→      last = out;
  1280→      if (!out.retryable) break;
  1281→    }
  1282→    await this.store.put(req.reference, last);
  1283→    return last;
  1284→  }
  1285→
  1286→  /** Look up an unknown delivery with the provider that took it. */
  1287→  async resolve(reference: string): Promise<VendOutcome> {
  1288→    const prior = await this.store.get(reference);
  1289→    if (!prior) return { status: 'failed', providerId: 'router', code: 'NOT_FOUND', message: 'no such delivery', retryable: false };
  1290→    if (prior.outcome.status !== 'unknown') return prior.outcome;
  1291→    const p = this.providers.find((x) => x.provider.id === prior.outcome.providerId)?.provider;
  1292→    if (!p) return prior.outcome;
  1293→    let out: VendOutcome;
  1294→    try { out = await p.status(reference); } catch (e) { return prior.outcome; }
  1295→    if (out.status !== 'unknown') await this.store.put(reference, out);
  1296→    return out.status === 'unknown' ? prior.outcome : out;
  1297→  }
  1298→
  1299→  async validateMeter(meterNumber: string, country: 'ZA' | 'MW', municipality?: string) {
  1300→    for (const { provider } of [...this.providers].sort((a, b) => a.priority - b.priority)) {
  1301→      if (provider.validateMeter && provider.supports('electricity', country)) return provider.validateMeter(meterNumber, country, municipality);
  1302→    }
  1303→    return { valid: false } as const;
  1304→  }
  1305→}
