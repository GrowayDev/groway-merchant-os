   387→### 13.5 Blockchain (Hyperledger Fabric) Chaincode
   388→
   389→```javascript
   390→// SAPE Chaincode — anchors product identity and custody events
   391→
   392→'use strict';
   393→
   394→const { Contract } = require('fabric-contract-api');
   395→
   396→class SAPEContract extends Contract {
   397→
   398→  // Anchor product identity
   399→  async anchorProduct(ctx, productIdentityId, gtin, batch, expiry, signature) {
   400→    const identity = {
   401→      productIdentityId,
   402→      gtin,
   403→      batch,
   404→      expiry,
   405→      signature,
   406→      timestamp: new Date().toISOString(),
   407→    };
   408→    await ctx.stub.putState(
   409→      `PRODUCT_${productIdentityId}`,
   410→      Buffer.from(JSON.stringify(identity))
   411→    );
   412→    return JSON.stringify(identity);
   413→  }
   414→
   415→  // Anchor custody event
   416→  async anchorCustodyEvent(ctx, eventId, productSerialId, eventType, location, actorId) {
   417→    const event = {
   418→      eventId,
   419→      productSerialId,
   420→      eventType,
   421→      location,
   422→      actorId,
   423→      timestamp: new Date().toISOString(),
   424→    };
   425→    await ctx.stub.putState(
   426→      `CUSTODY_${eventId}`,
   427→      Buffer.from(JSON.stringify(event))
   428→    );
   429→    return JSON.stringify(event);
   430→  }
   431→
   432→  // Verify product
   433→  async verifyProduct(ctx, productIdentityId) {
   434→    const identityBytes = await ctx.stub.getState(`PRODUCT_${productIdentityId}`);
   435→    if (!identityBytes || identityBytes.length === 0) {
   436→      throw new Error(`Product ${productIdentityId} not found`);
   437→    }
   438→    return identityBytes.toString();
   439→  }
   440→
   441→  // Query custody chain
   442→  async queryCustodyChain(ctx, productSerialId) {
   443→    const query = {
   444→      selector: { productSerialId: productSerialId },
   445→      sort: [{ timestamp: 'asc' }],
   446→    };
   447→    const iterator = await ctx.stub.getQueryResult(JSON.stringify(query));
   448→    const results = [];
   449→    while (true) {
   450→      const res = await iterator.next();
   451→      if (res.value && res.value.value.toString()) {
   452→        results.push(JSON.parse(res.value.value.toString('utf8')));
   453→      }
   454→      if (res.done) {
   455→        await iterator.close();
   456→        return JSON.stringify(results);
   457→      }
   458→    }
   459→  }
   460→}
   461→
   462→module.exports = SAPEContract;
   463→```

   593→class YocoAdapter implements PaymentAdapter {
   594→  private baseUrl = 'https://api.yoco.com/v1';
   595→  private secretKey: string;
   596→
   597→  async createPayment(request: CreatePaymentRequest): Promise<CreatePaymentResponse> {
   598→    const response = await fetch(`${this.baseUrl}/checkouts`, {
   599→      method: 'POST',
   600→      headers: {
   601→        'Authorization': `Bearer ${this.secretKey}`,
   602→        'Content-Type': 'application/json',
   603→      },
   604→      body: JSON.stringify({
   605→        amount: request.amount,
   606→        currency: 'ZAR',
   607→        successUrl: `${process.env.APP_URL}/payment/success`,
   608→        cancelUrl: `${process.env.APP_URL}/payment/cancel`,
   609→        failureUrl: `${process.env.APP_URL}/payment/failure`,
   610→        metadata: {
   611→          merchantId: request.merchantId,
   612→          reference: request.reference,
   613→        },
   614→      }),
   615→    });
   616→
   617→    const data = await response.json();
   618→    return {
   619→      paymentId: data.id,
   620→      provider: 'YOCO',
   621→      providerReference: data.id,
   622→      status: 'PENDING',
   623→      paymentUrl: data.redirectUrl,
   624→      expiresAt: new Date(data.expiresAt),
   625→    };
   626→  }
   627→
   628→  async verifyWebhook(payload: any, signature: string): Promise<boolean> {
   629→    const crypto = require('crypto');
   630→    const expectedSignature = crypto
   631→      .createHmac('sha256', process.env.YOCO_WEBHOOK_SECRET)
   632→      .update(JSON.stringify(payload))
   633→      .digest('hex');
   634→    return crypto.timingSafeEqual(
   635→      Buffer.from(signature),
   636→      Buffer.from(expectedSignature)
   637→    );
   638→  }
   639→
   640→  async refundPayment(paymentId: string, amount: number): Promise<RefundResponse> {
   641→    const response = await fetch(`${this.baseUrl}/refunds`, {
   642→      method: 'POST',
   643→      headers: {
   644→        'Authorization': `Bearer ${this.secretKey}`,
   645→        'Content-Type': 'application/json',
   646→      },
   647→      body: JSON.stringify({
   648→        chargeId: paymentId,
   649→        amount: amount,
   650→      }),
   651→    });
   652→    return response.json();
   653→  }
   654→}

   657→### 16.2 iKhokha
   658→
   659→```typescript
   660→// iKhokha Adapter (HMAC-SHA256 signature required)
   661→
   662→class iKhokhaAdapter implements PaymentAdapter {
   663→  private baseUrl = 'https://dev.ikhokha.com/api';
   664→  private appId: string;
   665→  private appSecret: string;
   666→
   667→  async createPayment(request: CreatePaymentRequest): Promise<CreatePaymentResponse> {
   668→    const timestamp = Date.now().toString();
   669→    const payload = {
   670→      amount: request.amount / 100,
   671→      currency: 'ZAR',
   672→      reference: request.reference,
   673→      merchantId: request.merchantId,
   674→      callbackUrl: `${process.env.APP_URL}/webhooks/ikhokha`,
   675→    };
   676→
   677→    const signature = this.generateSignature(payload, timestamp);
   678→
   679→    const response = await fetch(`${this.baseUrl}/pay`, {
   680→      method: 'POST',
   681→      headers: {
   682→        'IK-AppID': this.appId,
   683→        'IK-Timestamp': timestamp,
   684→        'IK-Signature': signature,
   685→        'Content-Type': 'application/json',
   686→      },
   687→      body: JSON.stringify(payload),
   688→    });
   689→
   690→    return response.json();
   691→  }
   692→
   693→  private generateSignature(payload: any, timestamp: string): string {
   694→    const crypto = require('crypto');
   695→    const payloadString = JSON.stringify(payload);
   696→    const signatureString = `${this.appId}${timestamp}${payloadString}`;
   697→    return crypto
   698→      .createHmac('sha256', this.appSecret)
   699→      .update(signatureString)
   700→      .digest('hex');
   701→  }
   702→}
