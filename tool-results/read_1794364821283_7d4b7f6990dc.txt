   104→// Margin Calculator Component
   105→function MarginCalculator() {
   106→  const [cost, setCost] = useState<number>(0)
   107→  const [sellPrice, setSellPrice] = useState<number>(0)
   108→  const [marginPercent, setMarginPercent] = useState<number>(0)
   109→  const [markupPercent, setMarkupPercent] = useState<number>(0)
   110→  const [calcMode, setCalcMode] = useState<'cost-sell' | 'cost-margin' | 'sell-margin'>('cost-sell')
   111→
   112→  const calculateFromCostSell = () => {
   113→    if (cost > 0 && sellPrice > 0) {
   114→      const margin = ((sellPrice - cost) / sellPrice) * 100
   115→      const markup = ((sellPrice - cost) / cost) * 100
   116→      setMarginPercent(Number(margin.toFixed(1)))
   117→      setMarkupPercent(Number(markup.toFixed(1)))
   118→    }
   119→  }
   120→
   121→  const calculateFromCostMargin = () => {
   122→    if (cost > 0 && marginPercent > 0) {
   123→      const sell = cost / (1 - marginPercent / 100)
   124→      const markup = ((sell - cost) / cost) * 100
   125→      setSellPrice(Number(sell.toFixed(2)))
   126→      setMarkupPercent(Number(markup.toFixed(1)))
   127→    }
   128→  }
   129→
   130→  const calculateFromSellMargin = () => {
   131→    if (sellPrice > 0 && marginPercent > 0) {
   132→      const c = sellPrice * (1 - marginPercent / 100)
   133→      const markup = ((sellPrice - c) / c) * 100
   134→      setCost(Number(c.toFixed(2)))
   135→      setMarkupPercent(Number(markup.toFixed(1)))
   136→    }
   137→  }
   138→
   139→  const handleCalculate = () => {
   140→    switch (calcMode) {
   141→      case 'cost-sell':
   142→        calculateFromCostSell()
   143→        break
   144→      case 'cost-margin':
   145→        calculateFromCostMargin()
   146→        break
   147→      case 'sell-margin':
   148→        calculateFromSellMargin()
   149→        break
   150→    }
   151→  }
   152→
   153→  return (
   154→    <Card>
   155→      <CardHeader className="pb-2">
   156→        <CardTitle className="text-sm flex items-center gap-2">
   157→          <Calculator className="h-4 w-4 text-[#00897B]" />
   158→          Margin Calculator
   159→        </CardTitle>
   160→      </CardHeader>
   161→      <CardContent className="space-y-4">
   162→        <div className="flex gap-2">
   163→          <Button
   164→            size="sm"
   165→            variant={calcMode === 'cost-sell' ? 'default' : 'outline'}
   166→            className={calcMode === 'cost-sell' ? 'bg-[#00897B] hover:bg-[#00695C]' : ''}
   167→            onClick={() => setCalcMode('cost-sell')}
   168→          >
   169→            Cost + Sell
   170→          </Button>
   171→          <Button
   172→            size="sm"
   173→            variant={calcMode === 'cost-margin' ? 'default' : 'outline'}
   174→            className={calcMode === 'cost-margin' ? 'bg-[#00897B] hover:bg-[#00695C]' : ''}
   175→            onClick={() => setCalcMode('cost-margin')}
   176→          >
   177→            Cost + Margin
   178→          </Button>
   179→          <Button
   180→            size="sm"
   181→            variant={calcMode === 'sell-margin' ? 'default' : 'outline'}
   182→            className={calcMode === 'sell-margin' ? 'bg-[#00897B] hover:bg-[#00695C]' : ''}
   183→            onClick={() => setCalcMode('sell-margin')}
   184→          >
   185→            Sell + Margin
   186→          </Button>
   187→        </div>
   188→
   189→        <div className="grid grid-cols-2 gap-3">
   190→          <div>
   191→            <label className="text-xs text-[#757575]">Cost Price (R)</label>
   192→            <Input
   193→              type="number"
   194→              value={cost || ''}
   195→              onChange={(e) => setCost(Number(e.target.value))}
   196→              placeholder="0.00"
   197→              disabled={calcMode === 'sell-margin'}
   198→            />
   199→          </div>
   200→          <div>
   201→            <label className="text-xs text-[#757575]">Selling Price (R)</label>
   202→            <Input
   203→              type="number"
   204→              value={sellPrice || ''}
   205→              onChange={(e) => setSellPrice(Number(e.target.value))}
   206→              placeholder="0.00"
   207→              disabled={calcMode === 'cost-margin'}
   208→            />
   209→          </div>
   210→          <div>
   211→            <label className="text-xs text-[#757575]">Margin %</label>
   212→            <Input
   213→              type="number"
   214→              value={marginPercent || ''}
   215→              onChange={(e) => setMarginPercent(Number(e.target.value))}
   216→              placeholder="0"
   217→              disabled={calcMode === 'cost-sell'}
   218→            />
   219→          </div>
   220→          <div>
   221→            <label className="text-xs text-[#757575]">Markup %</label>
   222→            <Input
   223→              type="number"
   224→              value={markupPercent || ''}
   225→              disabled
   226→              className="bg-gray-50"
   227→            />
   228→          </div>
   229→        </div>
   230→
   231→        <Button onClick={handleCalculate} className="w-full bg-[#00897B] hover:bg-[#00695C]">
   232→          Calculate
   233→        </Button>
   234→
   235→        {cost > 0 && sellPrice > 0 && (
   236→          <div className="p-3 bg-[#E0F2F1] rounded-lg">
   237→            <div className="grid grid-cols-2 gap-2 text-sm">
   238→              <div>
   239→                <span className="text-[#757575]">Profit per unit:</span>
   240→                <span className="font-bold text-[#00897B] ml-2">R{(sellPrice - cost).toFixed(2)}</span>
   241→              </div>
   242→              <div>
   243→                <span className="text-[#757575]">Margin:</span>
   244→                <span className="font-bold text-[#00897B] ml-2">{marginPercent}%</span>
   245→              </div>
   246→            </div>
   247→          </div>
   248→        )}
   249→      </CardContent>
   250→    </Card>
   251→  )
   252→}

   406→          costPrice: newCost,
   407→          sellingPrice: newSell,

   578→      {/* Margin Calculator */}
   579→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   580→        <MarginCalculator />
   581→        
   582→        {/* Quick Stats */}
   583→        <Card className="lg:col-span-2">
   584→          <CardHeader className="pb-2">
   585→            <CardTitle className="text-sm flex items-center gap-2">
   586→              <Target className="h-4 w-4 text-[#00897B]" />
   587→              Category Performance
   588→            </CardTitle>
   589→          </CardHeader>
   590→          <CardContent>
   591→            <div className="grid grid-cols-2 lg:grid-cols-4 gap-4">
   592→              {Object.entries(
   593→                pricingData.reduce((acc, item) => {
   594→                  if (!acc[item.category]) {

   598→                  acc[item.category].totalMargin += item.marginPercent

   611→            </div>
   612→          </CardContent>
   613→        </Card>
   614→      </div>
   615→    </div>
   616→  )
   617→
   618→  // Render Products Tab
   619→  const renderProducts = () => (
   620→    <div className="space-y-4">
   621→      {/* Filters */}
   622→      <div className="flex gap-3 flex-wrap">
   623→        <div className="relative flex-1 min-w-[200px]">
   624→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-[#757575]" />

   659→            <Edit2 className="h-4 w-4 mr-2" />
   660→            Update {selectedItems.length} Selected
   661→          </Button>
   662→        )}
   663→        <Button variant="outline" onClick={handleExport}>
   664→          <Download className="h-4 w-4 mr-2" />Export

   666→      </div>
   667→
   668→      {/* Products Table */}
   669→      <Card>
   670→        <ScrollArea className="h-[500px]">
   671→          <Table>
   672→            <TableHeader>
   673→              <TableRow>
   674→                <TableHead className="w-10">
   675→                  <input
   676→                    type="checkbox"
   677→                    checked={selectedItems.length === filteredItems.length && filteredItems.length > 0}
   678→                    onChange={handleSelectAll}
   679→                    className="rounded border-gray-300"
   680→                  />
   681→                </TableHead>
   682→                <TableHead>Product</TableHead>
   683→                <TableHead>Cost</TableHead>
   684→                <TableHead>Selling Price</TableHead>
   685→                <TableHead>Margin</TableHead>
   686→                <TableHead>Stock</TableHead>
   687→                <TableHead>Demand</TableHead>
   688→                <TableHead className="w-20">Actions</TableHead>
   689→              </TableRow>
   690→            </TableHeader>
   691→            <TableBody>
   692→              {filteredItems.map(item => (
   693→                <TableRow 
   694→                  key={item.id}
   695→                  className={selectedItems.includes(item.id) ? 'bg-[#E0F2F1]' : ''}
   696→                >
   697→                  <TableCell>
   698→                    <input
   699→                      type="checkbox"
   700→                      checked={selectedItems.includes(item.id)}
   701→                      onChange={() => handleSelectItem(item.id)}
   702→                      className="rounded border-gray-300"
   703→                    />
   704→                  </TableCell>
   705→                  <TableCell>
   706→                    <div>
   707→                      <p className="font-medium">{item.name}</p>
   708→                      <p className="text-xs text-[#757575]">{item.sku} • {item.category}</p>
   709→                    </div>
   710→                  </TableCell>
   711→                  <TableCell>
   712→                    {editItem?.id === item.id ? (
   713→                      <Input
   714→                        type="number"
   715→                        defaultValue={item.costPrice}
   716→                        className="w-24"
   717→                        onBlur={(e) => handleUpdatePrice(item, 'costPrice', Number(e.target.value))}
   718→                        autoFocus
   719→                      />
   720→                    ) : (
   721→                      <span>R{item.costPrice.toFixed(2)}</span>
   722→                    )}
   723→                  </TableCell>
   724→                  <TableCell>
   725→                    {editItem?.id === item.id ? (
   726→                      <Input
   727→                        type="number"
   728→                        defaultValue={item.sellingPrice}
   729→                        className="w-24"
   730→                        onBlur={(e) => handleUpdatePrice(item, 'sellingPrice', Number(e.target.value))}

   733→                      <span className="font-medium">R{item.sellingPrice.toFixed(2)}</span>

   735→                  </TableCell>

   737→                    <div className="flex items-center gap-2">
   738→                      <span className={`font-bold ${getMarginColor(item.marginPercent)}`}>
   739→                        {item.marginPercent.toFixed(1)}%
   740→                      </span>
   741→                      {item.marginPercent < 30 && (
   742→                        <AlertTriangle className="h-4 w-4 text-red-500" />

   744→                    </div>
   745→                    <p className="text-xs text-[#757575]">R{item.margin.toFixed(2)} profit</p>
   746→                  </TableCell>
   747→                  <TableCell>
   748→                    <span className={item.stock === 0 ? 'text-red-500' : item.stock < 20 ? 'text-orange-500' : ''}>

   772→    </div>
   773→  )
   774→
   775→  // Render History Tab
   776→  const renderHistory = () => {

   780→
   781→    return (

   815→                      </div>
   816→                      <div className="text-right">

   822→                </div>

   830→

   839→              Pricing Management
   840→            </h2>

   843→        </div>
   844→      </div>
   845→

   874→            <DialogDescription>
   875→              Update prices for {selectedItems.length} selected items
   876→            </DialogDescription>
   877→          </DialogHeader>
   878→          <div className="space-y-4">
   879→            <div>
   880→              <label className="text-sm font-medium">Update Field</label>
   881→              <Select 
   882→                value={bulkUpdate.field} 
   883→                onValueChange={(v) => setBulkUpdate(prev => ({ ...prev, field: v as 'sellingPrice' | 'costPrice' }))}

   885→                <SelectTrigger>

   887→                </SelectTrigger>
   888→                <SelectContent>
   889→                  <SelectItem value="sellingPrice">Selling Price</SelectItem>
   890→                  <SelectItem value="costPrice">Cost Price</SelectItem>
   891→                </SelectContent>
   892→              </Select>
   893→            </div>
   894→            <div>
   895→              <label className="text-sm font-medium">Operation</label>
   896→              <Select 
   897→                value={bulkUpdate.operation}

   899→              >
   900→                <SelectTrigger>
   901→                  <SelectValue />
   902→                </SelectTrigger>
   903→                <SelectContent>
   904→                  <SelectItem value="set">Set to value</SelectItem>
   905→                  <SelectItem value="increase">Increase by %</SelectItem>

   907→                  <SelectItem value="margin">Set margin %</SelectItem>
   908→                </SelectContent>

   910→            </div>
   911→            <div>
   912→              <label className="text-sm font-medium">
   913→                {bulkUpdate.operation === 'set' ? 'Value (R)' : 
   914→                 bulkUpdate.operation === 'margin' ? 'Target Margin %' : 'Percentage %'}
   915→              </label>
   916→              <Input
   917→                type="number"
   918→                value={bulkUpdate.value || ''}

   930→            </div>

   932→          <DialogFooter>
   933→            <Button variant="outline" onClick={() => setBulkUpdateOpen(false)}>Cancel</Button>

   941→      {/* Toast */}

   949→          <p className={`text-sm ${toast.type === 'success' ? 'text-green-600' : 'text-red-600'}`}>

   954→    </div>
