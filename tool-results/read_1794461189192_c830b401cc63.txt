     1→     1→/**
     2→     2→ * Repository ports for GroCore.
     3→     3→ *
     4→     4→ * Two implementations:
     5→     5→ *  - InMemoryRepository: for tests and quick local runs.
     6→     6→ *  - PrismaRepository: persists to the GroCore* tables.
     7→     7→ *
     8→     8→ * Each GroCore operation that needs persistence accepts a Repository.
     9→     9→ * This keeps the core logic pure and testable, and lets us swap the
    10→    10→ * persistence layer without changing business code.

   111→   111→  async getAccount(tenantId: string, code: string): Promise<LedgerAccount | null> {
   112→   112→    return this.accounts.get(`${tenantId}:${code}`) ?? null
   113→   113→  }
   114→   114→  async upsertAccount(account: LedgerAccount): Promise<LedgerAccount> {
   115→   115→    this.accounts.set(`${account.tenantId}:${account.code}`, account)
   116→   116→    return account
   117→   117→  }
   118→   118→  async listAccounts(tenantId: string): Promise<LedgerAccount[]> {
   119→   119→    return Array.from(this.accounts.values()).filter((a) => a.tenantId === tenantId)
   120→   120→  }
   121→   121→  async postEntry(entry: LedgerEntryInput, lines: Array<{ accountId: string; amountCents: bigint; currency: string }>): Promise<{ entry: LedgerEntry; lines: LedgerLine[] }> {
   122→   122→    const key = `${entry.tenantId}:${entry.entryId}`
   123→   123→    if (this.entries.has(key)) {
   124→   124→      return this.entries.get(key)!
   125→   125→    }
   126→   126→    const fullEntry: LedgerEntry = {
   127→   127→      ...entry,
   128→   128→      id: `e-${entry.entryId}`,
   129→   129→      postedAt: new Date(),
   130→   130→    }
   131→   131→    const fullLines: LedgerLine[] = lines.map((l, i) => ({
   132→   132→      id: `l-${entry.entryId}-${i}`,
   133→   133→      entryId: fullEntry.id,
   134→   134→      accountId: l.accountId,
   135→   135→      accountCode: '',
   136→   136→      amount: { cents: l.amountCents < 0n ? -l.amountCents : l.amountCents, currency: l.currency as any, signedCents: l.amountCents },
   137→   137→      currency: l.currency as any,
   138→   138→      amountCents: l.amountCents,
   139→   139→    }))
   140→   140→    this.entries.set(key, { entry: fullEntry, lines: fullLines })
   141→   141→    return { entry: fullEntry, lines: fullLines }
   142→   142→  }
   143→   143→  async getEntry(tenantId: string, entryId: string) {
   144→   144→    return this.entries.get(`${tenantId}:${entryId}`) ?? null
   145→   145→  }
   146→   146→  async listEntries(tenantId: string, limit = 100) {
   147→   147→    return Array.from(this.entries.values())
   148→   148→      .filter((e) => e.entry.tenantId === tenantId)
   149→   149→      .slice(0, limit)
   150→   150→  }
   151→   151→  async listLinesForAccount(_t: string, accountId: string) {
   152→   152→    const out: LedgerLine[] = []
   153→   153→    for (const { lines } of this.entries.values()) {
   154→   154→      out.push(...lines.filter((l) => l.accountId === accountId))
   155→   155→    }
   156→   156→    return out
   157→   157→  }
   158→   158→
   159→   159→  async getConsent(tenantId: string, subjectId: string, purpose: string) {
   160→   160→    return this.consents.get(`${tenantId}:${subjectId}:${purpose}`) ?? null
   161→   161→  }
   162→   162→  async listConsents(tenantId: string, subjectId?: string) {
   163→   163→    return Array.from(this.consents.values()).filter((c) =>
   164→   164→      c.tenantId === tenantId && (!subjectId || c.subjectId === subjectId),
   165→   165→    )

   175→   175→    return { id }
   176→   176→  }
   177→   177→  async listDecisions(tenantId: string, subjectId?: string) {

   186→   186→  }

   190→   190→  async getOpenAdvanceBySubject(tenantId: string, subjectId: string) {
   191→   191→    for (const a of this.advances.values()) {

   193→   193→    }
   194→   194→    return null
   195→   195→  }
   196→   196→  async listAdvances(tenantId: string, subjectId?: string) {

   198→   198→      a.tenantId === tenantId && (!subjectId || a.subjectId === subjectId),

   412→   412→export class PrismaRepository implements GroCoreRepository {
   413→   413→  async getAccount(tenantId: string, code: string) {
   414→   414→    const r = await db.groCoreAccount.findUnique({ where: { tenantId_code: { tenantId, code } } })
   415→   415→    return r ? rowToAccount(r) : null
   416→   416→  }
   417→   417→  async upsertAccount(account: LedgerAccount) {
   418→   418→    const r = await db.groCoreAccount.upsert({
   419→   419→      where: { tenantId_code: { tenantId: account.tenantId, code: account.code } },
   420→   420→      create: {
   421→   421→        id: account.id,
   422→   422→        tenantId: account.tenantId,
   423→   423→        code: account.code,
   424→   424→        name: account.name,
   425→   425→        kind: account.kind,
   426→   426→        subKind: account.subKind,
   427→   427→        currency: account.currency,
   428→   428→        active: account.active,
   429→   429→        metaJson: account.metaJson ?? null,
   430→   430→      },
   431→   431→      update: {
   432→   432→        name: account.name,
   433→   433→        kind: account.kind,
   434→   434→        subKind: account.subKind,
   435→   435→        active: account.active,
   436→   436→        metaJson: account.metaJson ?? null,
   437→   437→      },

   866→   866→/**
   867→   867→ * Get the active repository. Defaults to Prisma in non-test environments,
   868→   868→ * InMemory in test environments (set via GROCORE_REPO=memory).
   869→   869→ */
   870→   870→let cachedRepo: GroCoreRepository | null = null
   871→   871→export function getRepository(): GroCoreRepository {
   872→   872→  if (cachedRepo) return cachedRepo
   873→   873→  const useMemory = process.env.GROCORE_REPO === 'memory' || process.env.NODE_ENV === 'test'
   874→   874→  cachedRepo = useMemory ? new InMemoryRepository() : new PrismaRepository()
   875→   875→  return cachedRepo
   876→   876→}
