   325→// Get user by email - hybrid approach
   326→export async function getUserByEmail(email: string): Promise<HybridUser | null> {
   327→  const lowerEmail = email.toLowerCase()
   328→  
   329→  if (isProduction()) {
   330→    // Use in-memory storage for production
   331→    initializeMemoryStores()
   332→    return memoryUsers!.get(lowerEmail) || null
   333→  } else {
   334→    // Try database first for development
   335→    try {
   336→      const { db } = await import('@/lib/db')
   337→      const user = await db.user.findUnique({
   338→        where: { email: lowerEmail }
   339→      })
   340→      
   341→      if (user) {
   342→        return {
   343→          id: user.id,
   344→          email: user.email,
   345→          name: user.name,
   346→          phone: user.phone,
   347→          passwordHash: user.passwordHash || '',
   348→          pinHash: user.pinHash,
   349→          role: user.role,
   350→          emailVerified: user.emailVerified,
   351→          phoneVerified: user.phoneVerified,
   352→          lastLoginAt: user.lastLoginAt,
   353→          loginAttempts: user.loginAttempts,
   354→          lockedUntil: user.lockedUntil,
   355→          createdAt: user.createdAt,
   356→          updatedAt: user.updatedAt
   357→        }
   358→      }
   359→      
   360→      // Fallback to demo users if not in database
   361→      initializeMemoryStores()
   362→      return memoryUsers!.get(lowerEmail) || null
   363→    } catch {
   364→      // Database error, fallback to memory
   365→      initializeMemoryStores()
   366→      return memoryUsers!.get(lowerEmail) || null
   367→    }
   368→  }

   538→// Create session - hybrid approach
   539→export async function createSession(
   540→  userId: string,
   541→  token: string,
   542→  userAgent?: string,
   543→  ipAddress?: string
   544→): Promise<HybridSession> {
   545→  const session: HybridSession = {
   546→    id: `sess_${nanoid(12)}`,
   547→    userId,
   548→    token,
   549→    expiresAt: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000), // 7 days
   550→    userAgent: userAgent || null,
   551→    ipAddress: ipAddress || null,
   552→    createdAt: new Date()
   553→  }
   554→  
   555→  if (isProduction()) {
   556→    // Use in-memory storage for production
   557→    initializeMemoryStores()
   558→    memorySessions!.set(token, session)
   559→    return session
   560→  } else {
   561→    // Try database first for development
   562→    try {
   563→      const { db } = await import('@/lib/db')
   564→      
   565→      await db.session.create({
   566→        data: {
   567→          id: session.id,
   568→          userId,
   569→          token,
   570→          expiresAt: session.expiresAt,
   571→          userAgent,
   572→          ipAddress
   573→        }
   574→      })
   575→      
   576→      return session
   577→    } catch {
   578→      // Database error, use memory
   579→      initializeMemoryStores()
   580→      memorySessions!.set(token, session)
   581→      return session
   582→    }
   583→  }
   584→}

   586→// Update user login info - hybrid approach

   590→    initializeMemoryStores()
   591→

   594→      if (user.id === userId) {
   595→        user.lastLoginAt = new Date()
   596→        user.loginAttempts = 0
   597→        user.lockedUntil = null
   598→        user.updatedAt = new Date()

   601→      }
   602→    }
   603→  } else {
   604→    // Try database first for development

   611→          loginAttempts: 0,
   612→          lockedUntil: null,
   613→          lastLoginAt: new Date()
   614→        }
   615→      })
   616→    } catch {
   617→      // Database error, update in memory
   618→      initializeMemoryStores()
   619→      
   620→      for (const [email, user] of memoryUsers!.entries()) {
   621→        if (user.id === userId) {
