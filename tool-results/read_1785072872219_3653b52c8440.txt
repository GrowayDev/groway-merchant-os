    72→// Simple QR Code component for tickets
    73→function TicketQRCode({ data, size = 120 }: { data: string; size?: number }) {
    74→  // Generate a deterministic pattern based on the data string
    75→  const generatePattern = (str: string) => {
    76→    const pattern: boolean[][] = [];
    77→    let hash = 0;
    78→    for (let i = 0; i < str.length; i++) {
    79→      hash = ((hash << 5) - hash) + str.charCodeAt(i);
    80→      hash = hash & hash;
    81→    }
    82→    
    83→    // Create a 21x21 grid (small QR code)
    84→    const gridSize = 21;
    85→    for (let row = 0; row < gridSize; row++) {
    86→      pattern[row] = [];
    87→      for (let col = 0; col < gridSize; col++) {
    88→        // Add position patterns (corners)
    89→        const isCorner = 
    90→          (row < 7 && col < 7) || 
    91→          (row < 7 && col >= gridSize - 7) || 
    92→          (row >= gridSize - 7 && col < 7);
    93→        
    94→        if (isCorner) {
    95→          // Position pattern
    96→          const inCornerOuter = row < 7 && col < 7;
    97→          const inCornerTR = row < 7 && col >= gridSize - 7;
    98→          const inCornerBL = row >= gridSize - 7 && col < 7;
    99→          
   100→          let r = row, c = col;
   101→          if (inCornerTR) { r = row; c = col - (gridSize - 7); }
   102→          if (inCornerBL) { r = row - (gridSize - 7); c = col; }
   103→          
   104→          if (r === 0 || r === 6 || c === 0 || c === 6) {
   105→            pattern[row][col] = true;
   106→          } else if (r >= 2 && r <= 4 && c >= 2 && c <= 4) {
   107→            pattern[row][col] = true;
   108→          } else {
   109→            pattern[row][col] = false;
   110→          }
   111→        } else {
   112→          // Data area - use hash to determine
   113→          const seed = (hash + row * gridSize + col) % 100;
   114→          pattern[row][col] = seed < 45;
   115→        }
   116→      }
   117→    }
   118→    return pattern;
   119→  };
   120→
   121→  const pattern = generatePattern(data || 'default');
   122→  const cellSize = size / 21;
   123→
   124→  return (
   125→    <div className="bg-white p-2 rounded-lg inline-block">
   126→      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
   127→        {pattern.map((row, rowIndex) =>
   128→          row.map((cell, colIndex) => (
   129→            <rect
   130→              key={`${rowIndex}-${colIndex}`}
   131→              x={colIndex * cellSize}
   132→              y={rowIndex * cellSize}
   133→              width={cellSize}
   134→              height={cellSize}
   135→              fill={cell ? '#000000' : '#ffffff'}
   136→            />
   137→          ))
   138→        )}
   139→      </svg>
   140→    </div>
   141→  );
   142→}

   144→// IATA Aztec Barcode Component for Boarding Passes
   145→function AztecBarcode({ 
   146→  data, 
   147→  size = 150 
   148→}: { 
   149→  data: {
   150→    passengerName: string;
   151→    pnr: string;
   152→    from: string;
   153→    to: string;
   154→    flightNumber: string;
   155→    date: string;
   156→    seat?: string;
   157→    class: string;
   158→    seq?: string;
   159→  }; 
   160→  size?: number;
   161→}) {
   162→  // Generate Aztec-style pattern (IATA BCBP standard)
   163→  // Aztec has a central finder pattern (bullseye) and data layers around it
   164→  const generateAztecPattern = () => {
   165→    const gridSize = 23; // Standard Aztec compact
   166→    const pattern: boolean[][] = [];
   167→    const center = Math.floor(gridSize / 2);
   168→    
   169→    // Create hash from data
   170→    let hash = 0;
   171→    const dataStr = `${data.passengerName}${data.pnr}${data.flightNumber}${data.from}${data.to}${data.date}`;
   172→    for (let i = 0; i < dataStr.length; i++) {
   173→      hash = ((hash << 5) - hash) + dataStr.charCodeAt(i);
   174→      hash = hash & hash;
   175→    }
   176→
   177→    for (let row = 0; row < gridSize; row++) {
   178→      pattern[row] = [];
   179→      for (let col = 0; col < gridSize; col++) {
   180→        const distFromCenter = Math.max(Math.abs(row - center), Math.abs(col - center));
   181→        
   182→        // Central bullseye finder pattern (Aztec style)
   183→        if (distFromCenter <= 4) {
   184→          // Concentric rings
   185→          if (distFromCenter <= 1) {
   186→            pattern[row][col] = true; // Center solid
   187→          } else if (distFromCenter === 2) {
   188→            pattern[row][col] = false; // White ring
   189→          } else if (distFromCenter === 3) {
   190→            pattern[row][col] = true; // Black ring
   191→          } else {
   192→            pattern[row][col] = false; // White ring
   193→          }
   194→        } else {
   195→          // Data area - deterministic based on position and hash
   196→          const seed = (hash + row * gridSize + col + distFromCenter * 7) % 100;
   197→          pattern[row][col] = seed < 48;
   198→        }
   199→      }
   200→    }
   201→    return pattern;
   202→  };
   203→
   204→  const pattern = generateAztecPattern();
   205→  const cellSize = size / 23;
   206→
   207→  // Generate IATA BCBP data string
   208→  const bcbpData = `M${data.passengerName.toUpperCase().substring(0, 20).padEnd(20)}${data.pnr.padEnd(7)}${data.from}${data.to}${data.flightNumber.padEnd(6)}${data.date}${data.class}${data.seat || '000'}${data.seq || '001'}`;
   209→
   210→  return (
   211→    <div className="bg-white p-3 rounded-lg inline-block">
   212→      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
   213→        {pattern.map((row, rowIndex) =>
   214→          row.map((cell, colIndex) => (
   215→            <rect
   216→              key={`${rowIndex}-${colIndex}`}
   217→              x={colIndex * cellSize}
   218→              y={rowIndex * cellSize}
   219→              width={cellSize}
   220→              height={cellSize}
   221→              fill={cell ? '#000000' : '#ffffff'}
   222→            />
   223→          ))
   224→        )}
   225→      </svg>
   226→      <p className="text-[6px] text-gray-400 mt-1 font-mono text-center break-all max-w-[150px]">{bcbpData.substring(0, 40)}...</p>
   227→    </div>
   228→  );
   229→}
