     1→/**
     2→ * Repository ports for GroCore.
     3→ *
     4→ * Two implementations:
     5→ *  - InMemoryRepository: for tests and quick local runs.
     6→ *  - PrismaRepository: persists to the GroCore* tables.
     7→ *
     8→ * Each GroCore operation that needs persistence accepts a Repository.
     9→ * This keeps the core logic pure and testable, and lets us swap the
    10→ * persistence layer without changing business code.
    11→ */

    45→  // Consent

    49→

    52→    tenantId: string
    53→    subjectId: string
    54→    subjectType: string
    55→    product: ProductCode
    56→    amountCents: bigint
    57→    currency: string
    58→    outcome: string

    60→    reasonCodes: string[]
    61→    counterfactual: string | null
    62→    traceJson: string
    63→    policyVersion: string
    64→    modelVersion: string
    65→  }): Promise<{ id: string }>
    66→  listDecisions(tenantId: string, subjectId?: string): Promise<any[]>
    67→
    68→  // Advances
    69→  saveAdvance(advance: TopupAdvance): Promise<TopupAdvance>
    70→  getAdvance(tenantId: string, advanceId: string): Promise<TopupAdvance | null>
    71→  getOpenAdvanceBySubject(tenantId: string, subjectId: string): Promise<TopupAdvance | null>
    72→  listAdvances(tenantId: string, subjectId?: string): Promise<TopupAdvance[]>
    73→
    74→  // Vouchers

    78→
    79→  // Capital sources

    81→  listCapitalSources(tenantId: string): Promise<CapitalSource[]>

    83→
    84→  // GoPay

    93→  listDeviceContracts(tenantId: string, customerId?: string): Promise<DeviceContract[]>
    94→}

    97→// In-memory implementation (for tests)

    99→export class InMemoryRepository implements GroCoreRepository {
   100→  private accounts = new Map<string, LedgerAccount>()
   101→  private entries = new Map<string, { entry: LedgerEntry; lines: LedgerLine[] }>()
   102→  private consents = new Map<string, ConsentRecord>()
   103→  private decisions: any[] = []
   104→  private advances = new Map<string, TopupAdvance>()
   105→  private vouchers = new Map<string, Voucher>()
   106→  private capitalSources = new Map<string, CapitalSource>()
   107→  private goPayCaptures = new Map<string, GoPayCapture>()
   108→  private goPaySettlements = new Map<string, GoPaySettlement>()
   109→  private deviceContracts = new Map<string, DeviceContract>()
   110→
   111→  async getAccount(tenantId: string, code: string): Promise<LedgerAccount | null> {
   112→    return this.accounts.get(`${tenantId}:${code}`) ?? null
   113→  }
   114→  async upsertAccount(account: LedgerAccount): Promise<LedgerAccount> {
   115→    this.accounts.set(`${account.tenantId}:${account.code}`, account)
   116→    return account
   117→  }
   118→  async listAccounts(tenantId: string): Promise<LedgerAccount[]> {
   119→    return Array.from(this.accounts.values()).filter((a) => a.tenantId === tenantId)
   120→  }
   121→  async postEntry(entry: LedgerEntryInput, lines: Array<{ accountId: string; amountCents: bigint; currency: string }>): Promise<{ entry: LedgerEntry; lines: LedgerLine[] }> {
   122→    const key = `${entry.tenantId}:${entry.entryId}`
   123→    if (this.entries.has(key)) {
   124→      return this.entries.get(key)!
   125→    }
   126→    const fullEntry: LedgerEntry = {
   127→      ...entry,
   128→      id: `e-${entry.entryId}`,
   129→      postedAt: new Date(),
   130→    }
   131→    const fullLines: LedgerLine[] = lines.map((l, i) => ({
   132→      id: `l-${entry.entryId}-${i}`,
   133→      entryId: fullEntry.id,
   134→      accountId: l.accountId,
   135→      accountCode: '',
   136→      amount: { cents: l.amountCents < 0n ? -l.amountCents : l.amountCents, currency: l.currency as any, signedCents: l.amountCents },
   137→      currency: l.currency as any,
   138→      amountCents: l.amountCents,
   139→    }))
   140→    this.entries.set(key, { entry: fullEntry, lines: fullLines })
   141→    return { entry: fullEntry, lines: fullLines }
   142→  }
   143→  async getEntry(tenantId: string, entryId: string) {
   144→    return this.entries.get(`${tenantId}:${entryId}`) ?? null
   145→  }
   146→  async listEntries(tenantId: string, limit = 100) {
   147→    return Array.from(this.entries.values())
   148→      .filter((e) => e.entry.tenantId === tenantId)
   149→      .slice(0, limit)
   150→  }
   151→  async listLinesForAccount(_t: string, accountId: string) {
   152→    const out: LedgerLine[] = []
   153→    for (const { lines } of this.entries.values()) {
   154→      out.push(...lines.filter((l) => l.accountId === accountId))
   155→    }
   156→    return out
   157→  }
   158→
   159→  async getConsent(tenantId: string, subjectId: string, purpose: string) {
   160→    return this.consents.get(`${tenantId}:${subjectId}:${purpose}`) ?? null
   161→  }
   162→  async listConsents(tenantId: string, subjectId?: string) {
   163→    return Array.from(this.consents.values()).filter((c) =>
   164→      c.tenantId === tenantId && (!subjectId || c.subjectId === subjectId),
   165→    )
   166→  }
   167→  async saveConsent(record: ConsentRecord) {
   168→    this.consents.set(`${record.tenantId}:${record.subjectId}:${record.purpose}`, record)
   169→    return record
   170→  }
   171→
   172→  async saveDecision(decision: any) {
   173→    const id = `d-${this.decisions.length + 1}`
   174→    this.decisions.push({ id, ...decision, createdAt: new Date() })
   175→    return { id }
   176→  }
   177→  async listDecisions(tenantId: string, subjectId?: string) {
   178→    return this.decisions.filter((d) =>
   179→      d.tenantId === tenantId && (!subjectId || d.subjectId === subjectId),
   180→    )
   181→  }
   182→
   183→  async saveAdvance(advance: TopupAdvance) {
   184→    this.advances.set(`${advance.tenantId}:${advance.id}`, advance)
   185→    return advance
   186→  }
   187→  async getAdvance(tenantId: string, id: string) {
   188→    return this.advances.get(`${tenantId}:${id}`) ?? null
   189→  }
   190→  async getOpenAdvanceBySubject(tenantId: string, subjectId: string) {
   191→    for (const a of this.advances.values()) {
   192→      if (a.tenantId === tenantId && a.subjectId === subjectId && a.status === 'open') return a
   193→    }
   194→    return null
   195→  }
   196→  async listAdvances(tenantId: string, subjectId?: string) {
   197→    return Array.from(this.advances.values()).filter((a) =>
   198→      a.tenantId === tenantId && (!subjectId || a.subjectId === subjectId),
   199→    )
   200→  }
   201→
   202→  async saveVoucher(voucher: Voucher) {
   203→    this.vouchers.set(`${voucher.tenantId}:${voucher.serial}`, voucher)
   204→    return voucher
   205→  }
   206→  async getVoucher(tenantId: string, serial: string) {
   207→    return this.vouchers.get(`${tenantId}:${serial}`) ?? null
   208→  }
   209→  async listVouchers(tenantId: string, filter?: { product?: string; status?: string; recipientMsisdn?: string }) {
   210→    return Array.from(this.vouchers.values()).filter((v) =>
   211→      v.tenantId === tenantId &&
   212→      (!filter?.product || v.product === filter.product) &&
   213→      (!filter?.status || v.status === filter.status) &&
   214→      (!filter?.recipientMsisdn || v.recipientMsisdn === filter.recipientMsisdn),
   215→    )
   216→  }
   217→
   218→  async saveCapitalSource(source: CapitalSource) {
   219→    this.capitalSources.set(`${source.tenantId}:${source.id}`, source)
   220→    return source
   221→  }
   222→  async listCapitalSources(tenantId: string) {
   223→    return Array.from(this.capitalSources.values()).filter((s) => s.tenantId === tenantId)
   224→  }
   225→  async getCapitalSource(tenantId: string, id: string) {
   226→    return this.capitalSources.get(`${tenantId}:${id}`) ?? null
   227→  }
   228→
   229→  async saveGoPayCapture(capture: GoPayCapture) {
   230→    this.goPayCaptures.set(`${capture.tenantId}:${capture.id}`, capture)
   231→    return capture
   232→  }
   233→  async listGoPayCaptures(tenantId: string, merchantId?: string) {
   234→    return Array.from(this.goPayCaptures.values()).filter((c) =>
   235→      c.tenantId === tenantId && (!merchantId || c.merchantId === merchantId),
   236→    )
   237→  }
   238→  async saveGoPaySettlement(settlement: GoPaySettlement) {
   239→    this.goPaySettlements.set(`${settlement.tenantId}:${settlement.id}`, settlement)
   240→    return settlement
   241→  }
   242→  async listGoPaySettlements(tenantId: string, merchantId?: string) {
   243→    return Array.from(this.goPaySettlements.values()).filter((s) =>
   244→      s.tenantId === tenantId && (!merchantId || s.merchantId === merchantId),
   245→    )
   246→  }
   247→
   248→  async saveDeviceContract(contract: DeviceContract) {
   249→    this.deviceContracts.set(`${contract.tenantId}:${contract.id}`, contract)
   250→    return contract
   251→  }
   252→  async getDeviceContract(tenantId: string, id: string) {
   253→    return this.deviceContracts.get(`${tenantId}:${id}`) ?? null
   254→  }
   255→  async listDeviceContracts(tenantId: string, customerId?: string) {
   256→    return Array.from(this.deviceContracts.values()).filter((c) =>
   257→      c.tenantId === tenantId && (!customerId || c.customerId === customerId),
   258→    )
   259→  }
   260→}

   412→export class PrismaRepository implements GroCoreRepository {
   413→  async getAccount(tenantId: string, code: string) {
   414→    const r = await db.groCoreAccount.findUnique({ where: { tenantId_code: { tenantId, code } } })
   415→    return r ? rowToAccount(r) : null
   416→  }
   417→  async upsertAccount(account: LedgerAccount) {
   418→    const r = await db.groCoreAccount.upsert({
   419→      where: { tenantId_code: { tenantId: account.tenantId, code: account.code } },
   420→      create: {
   421→        id: account.id,
   422→        tenantId: account.tenantId,
   423→        code: account.code,
   424→        name: account.name,
   425→        kind: account.kind,
   426→        subKind: account.subKind,
   427→        currency: account.currency,
   428→        active: account.active,
   429→        metaJson: account.metaJson ?? null,
   430→      },
   431→      update: {
   432→        name: account.name,
   433→        kind: account.kind,
   434→        subKind: account.subKind,
   435→        active: account.active,
   436→        metaJson: account.metaJson ?? null,
   437→      },
   438→    })
   439→    return rowToAccount(r)
   440→  }
   441→  async listAccounts(tenantId: string) {
   442→    const rows = await db.groCoreAccount.findMany({ where: { tenantId } })
   443→    return rows.map(rowToAccount)
   444→  }
   445→  async postEntry(entry: LedgerEntryInput, lines: Array<{ accountId: string; amountCents: bigint; currency: string }>) {
   446→    const existing = await db.groCoreEntry.findUnique({
   447→      where: { tenantId_entryId: { tenantId: entry.tenantId, entryId: entry.entryId } },
   448→      include: { lines: true },
   449→    })
   450→    if (existing) {
   451→      return {
   452→        entry: {
   453→          id: existing.id,
   454→          ...entry,
   455→          postedAt: existing.postedAt,
   456→        },
   457→        lines: existing.lines.map((l) => ({
   458→          id: l.id,
   459→          entryId: existing.id,
   460→          accountId: l.accountId,
   461→          accountCode: '',
   462→          amount: { cents: l.amountCents < 0n ? -l.amountCents : l.amountCents, currency: l.currency as any, signedCents: l.amountCents },
   463→          currency: l.currency as any,
   464→          amountCents: l.amountCents,
   465→        })),
   466→      }
   467→    }
   468→    const created = await db.groCoreEntry.create({
   469→      data: {
   470→        tenantId: entry.tenantId,
   471→        entryId: entry.entryId,
   472→        sourceType: entry.sourceType,
   473→        sourceId: entry.sourceId,
   474→        description: entry.description,
   475→        currency: lines[0]?.currency ?? 'ZAR',
   476→        lines: {
   477→          create: lines.map((l) => ({
   478→            tenantId: entry.tenantId,
   479→            accountId: l.accountId,
   480→            amountCents: l.amountCents,
   481→            currency: l.currency,
   482→          })),
   483→        },
   484→      },
   485→      include: { lines: true },
   486→    })
   487→    return {
   488→      entry: {
   489→        id: created.id,
   490→        ...entry,
   491→        postedAt: created.postedAt,
   492→      },
   493→      lines: created.lines.map((l) => ({
   494→        id: l.id,
   495→        entryId: created.id,
   496→        accountId: l.accountId,
   497→        accountCode: '',
   498→        amount: { cents: l.amountCents < 0n ? -l.amountCents : l.amountCents, currency: l.currency as any, signedCents: l.amountCents },
   499→        currency: l.currency as any,
   500→        amountCents: l.amountCents,
   501→      })),
   502→    }
   503→  }
   504→  async getEntry(tenantId: string, entryId: string) {
   505→    const r = await db.groCoreEntry.findUnique({
   506→      where: { tenantId_entryId: { tenantId, entryId } },
   507→      include: { lines: true },
   508→    })
   509→    if (!r) return null
   510→    return {
   511→      entry: { id: r.id, tenantId: r.tenantId, entryId: r.entryId, sourceType: r.sourceType, sourceId: r.sourceId, description: r.description, lines: [], postedAt: r.postedAt },
   512→      lines: r.lines.map((l) => ({
   513→        id: l.id,
   514→        entryId: r.id,
   515→        accountId: l.accountId,
   516→        accountCode: '',
   517→        amount: { cents: l.amountCents < 0n ? -l.amountCents : l.amountCents, currency: l.currency as any, signedCents: l.amountCents },
   518→        currency: l.currency as any,
   519→        amountCents: l.amountCents,
   520→      })),
   521→    }
   522→  }
   523→  async listEntries(tenantId: string, limit = 100) {
   524→    const rows = await db.groCoreEntry.findMany({
   525→      where: { tenantId },

   866→/**
   867→ * Get the active repository. Defaults to Prisma in non-test environments,
   868→ * InMemory in test environments (set via GROCORE_REPO=memory).
   869→ */
   870→let cachedRepo: GroCoreRepository | null = null
   871→export function getRepository(): GroCoreRepository {
   872→  if (cachedRepo) return cachedRepo
   873→  const useMemory = process.env.GROCORE_REPO === 'memory' || process.env.NODE_ENV === 'test'
   874→  cachedRepo = useMemory ? new InMemoryRepository() : new PrismaRepository()
   875→  return cachedRepo
   876→}
   877→
   878→export function setRepository(repo: GroCoreRepository): void {
   879→  cachedRepo = repo
   880→}
