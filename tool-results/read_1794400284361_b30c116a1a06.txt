    71→   670→   670→  return `${neg ? '-' : ''}R${whole}.${(abs % 100n).toString().padStart(2, '0')}`;

   186→   785→   785→

   193→   792→   792→  menuMinor: bigint[];
   194→   793→   793→  fee: FeeModel;
   195→   794→   794→  /** Hard ceiling on the fee for one advance, set from the rules that apply to the product. Checked at quote time. */
   196→   795→   795→  maxFeeMinor?: bigint;
   197→   796→   796→  recovery: RecoveryRule;
   198→   797→   797→  minTenureDays: number;
   199→   798→   798→  /** Eligibility: at least minMonths of the last 3 calendar months with recharges totalling minMonthlyRechargeMinor. */
   200→   799→   799→  minMonths: number;
   201→   800→   800→  minMonthlyRechargeMinor: bigint;
   202→   801→   801→  /** Largest advance as a share of average monthly recharge, in basis points. */
   203→   802→   802→  limitShareBps: number;
   204→   803→   803→  /** After this many days unrecovered, the overdue action applies. */

   209→   808→   808→export function quoteFee(p: Product, principal: bigint): bigint {
   210→   809→   809→  if (principal <= 0n) throw new RangeError('principal must be positive');
   211→   810→   810→  let fee: bigint;
   212→   811→   811→  switch (p.fee.kind) {
   213→   812→   812→    case 'flat': fee = p.fee.feeMinor; break;
   214→   813→   813→    case 'per_block': fee = ((principal + p.fee.blockMinor - 1n) / p.fee.blockMinor) * p.fee.feePerBlockMinor; break;
   215→   814→   814→    case 'percent_bps': fee = mulBps(principal, BigInt(p.fee.bps)); break;
   216→   815→   815→  }
   217→   816→   816→  if (p.maxFeeMinor !== undefined && fee > p.maxFeeMinor) {
   218→   817→   817→    throw new RangeError(`fee ${fee} exceeds the product ceiling ${p.maxFeeMinor}`);
   219→   818→   818→  }
   220→   819→   819→  return fee;

   236→   835→   835→  if (h.arrearsMinor > 0n) reasons.push('ARREARS_PRESENT');
   237→   836→   836→  if (h.tenureDays < p.minTenureDays) reasons.push('TENURE_TOO_SHORT');
   238→   837→   837→
   239→   838→   838→  const since = new Date(now.getTime() - 90 * 86400000);
   240→   839→   839→  const byMonth = new Map<string, bigint>();
   241→   840→   840→  let total = 0n;

   244→   843→   843→    const k = `${r.at.getUTCFullYear()}-${r.at.getUTCMonth()}`;

   254→   853→   853→  const maxMinor = eligibleMenu.length ? eligibleMenu.reduce((a, b) => (a > b ? a : b)) : 0n;
