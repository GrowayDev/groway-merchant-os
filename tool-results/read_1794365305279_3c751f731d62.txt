   134→export function SalesReportView() {
   135→  const [data, setData] = useState<SalesReportData | null>(null)

   141→  const [selectedMonth, setSelectedMonth] = useState(new Date().getMonth())
   142→
   143→  const fetchData = useCallback(async () => {
   144→    setIsLoading(true)
   145→    setError(null)
   146→    
   147→    try {
   148→      const params = new URLSearchParams({
   149→        merchantId: DEMO_MERCHANT_ID,
   150→        period,
   151→        year: selectedYear.toString(),
   152→      })
   153→      
   154→      if (period === 'quarterly') {
   155→        params.set('quarter', selectedQuarter.toString())
   156→      }
   157→      
   158→      const response = await fetch(`/merchant-os/api/reports/sales?${params}`)
   159→      
   160→      if (!response.ok) {
   161→        throw new Error('Failed to fetch sales report')
   162→      }
   163→      
   164→      const result = await response.json()
   165→      
   166→      if (result.success && result.data) {
   167→        setData(result.data)
   168→      } else {
   169→        throw new Error(result.error || 'Failed to load data')
   170→      }
   171→    } catch (err) {
   172→      console.error('Error fetching sales report:', err)
   173→      setError(err instanceof Error ? err.message : 'Failed to load sales report')
   174→      // Use fallback mock data if API fails
   175→      setData(generateMockData(period, selectedYear, selectedQuarter))
   176→    } finally {
   177→      setIsLoading(false)
   178→    }
   179→  }, [period, selectedYear, selectedQuarter])

   243→      hourlyBreakdown.push({
   244→        hour: `${h.toString().padStart(2, '0')}:00`,

   304→        ? monthlyBreakdown.reduce((sum, m) => sum + m.transactions, 0)
   305→        : periodType === 'weekly'
   306→          ? dailyBreakdown.reduce((sum, d) => sum + d.transactions, 0)
   307→          : 35 + Math.floor(Math.random() * 20)
   308→

   310→      period: periodType,
   311→      periodStart: periodStart.toISOString(),
   312→      periodEnd: periodEnd.toISOString(),
   313→      year,
   314→      quarter: periodType === 'quarterly' ? quarter : undefined,
   315→      metrics: {
   316→        totalSales,
   317→        transactionCount,
   318→        avgTransaction: Math.round(totalSales / transactionCount),
   319→        growth: -10 + Math.floor(Math.random() * 25)
   320→      },
   321→      hourlyBreakdown,
   322→      paymentMethods: [
   323→        { method: 'card', amount: Math.round(totalSales * 0.55), count: Math.round(transactionCount * 0.55), percentage: 55 },
   324→        { method: 'cash', amount: Math.round(totalSales * 0.25), count: Math.round(transactionCount * 0.25), percentage: 25 },
   325→        { method: 'qr', amount: Math.round(totalSales * 0.12), count: Math.round(transactionCount * 0.12), percentage: 12 },
   326→        { method: 'wallet', amount: Math.round(totalSales * 0.08), count: Math.round(transactionCount * 0.08), percentage: 8 },
   327→      ],
   328→      topProducts: [
   329→        { id: 'prod-1', name: 'Burger Combo', sold: 168, revenue: 15103.20 },
   330→        { id: 'prod-2', name: 'Chicken Wings (12pc)', sold: 126, revenue: 12587.40 },
   331→        { id: 'prod-3', name: 'Beef Burger Deluxe', sold: 105, revenue: 7339.50 },
   332→        { id: 'prod-4', name: 'Coca-Cola 500ml', sold: 315, revenue: 5953.50 },
   333→        { id: 'prod-5', name: 'Milkshake Chocolate', sold: 84, revenue: 2931.60 },
   334→      ],
   335→      topCategories: [
   336→        { category: 'Combos', revenue: Math.round(totalSales * 0.35), percentage: 35 },
   337→        { category: 'Burgers', revenue: Math.round(totalSales * 0.25), percentage: 25 },
   338→        { category: 'Beverages', revenue: Math.round(totalSales * 0.20), percentage: 20 },
   339→        { category: 'Sides', revenue: Math.round(totalSales * 0.12), percentage: 12 },
   340→        { category: 'Desserts', revenue: Math.round(totalSales * 0.08), percentage: 8 },
   341→      ],
   342→      dailyBreakdown,
   343→      monthlyBreakdown,
   344→      quarterlyBreakdown,
   345→      comparison: {
   346→        current: totalSales,
   347→        previous: Math.round(totalSales * (0.85 + Math.random() * 0.2)),
   348→        change: -10 + Math.floor(Math.random() * 25)
   349→      },
   350→      generatedAt: new Date().toISOString()

   352→  }

   357→        if (selectedQuarter === 1) {
   358→          setSelectedQuarter(4)
   359→          setSelectedYear(selectedYear - 1)
   360→        } else {
   361→          setSelectedQuarter(selectedQuarter - 1)

   363→      } else {
   364→        if (selectedQuarter === 4) {
   365→          setSelectedQuarter(1)
   366→          setSelectedYear(selectedYear + 1)
   367→        } else {
   368→          setSelectedQuarter(selectedQuarter + 1)
   369→        }
   370→      }
   371→    } else if (period === 'yearly') {
   372→      setSelectedYear(selectedYear + (direction === 'next' ? 1 : -1))
   373→    }
   374→  }
   375→
   376→  const getPeriodLabel = () => {
   377→    switch (period) {
   378→      case 'daily':
   379→        return 'Today'
   380→      case 'weekly':
   381→        return 'This Week'
   382→      case 'monthly':
   383→        return `${monthNames[selectedMonth]} ${selectedYear}`
   384→      case 'quarterly':
   385→        return `Q${selectedQuarter} ${selectedYear}`
   386→      case 'yearly':
   387→        return `Year ${selectedYear}`
   388→      default:
   389→        return ''
   390→    }
   391→  }

   399→        </div>

   407→        <div className="text-center">
   408→          <AlertCircle className="h-8 w-8 mx-auto mb-4 text-red-500" />
   409→          <p className="text-[#757575]">Failed to load sales report</p>
   410→          <Button variant="outline" size="sm" onClick={fetchData} className="mt-4">
   411→            <RefreshCw className="h-4 w-4 mr-2" />
   412→            Retry
   413→          </Button>
   414→        </div>
   415→      </div>

   420→
   421→  return (

   423→      <div className="p-6 space-y-6">
   424→        {/* Header */}
   425→        <div className="flex items-center justify-between flex-wrap gap-4">
   426→          <div>
   427→            <h1 className="text-2xl font-bold text-[#1A1A1A] flex items-center gap-2">
   428→              <BarChart3 className="h-6 w-6 text-[#00897B]" />
   429→              Sales Report
   430→            </h1>
   431→            <p className="text-sm text-[#757575]">
   432→              Generated on {new Date(data.generatedAt).toLocaleString('en-ZA')}

   436→            <Button variant="outline" size="sm" onClick={fetchData}>
   437→              <RefreshCw className="h-4 w-4 mr-2" />
   438→              Refresh
   439→            </Button>

   442→              Print
   443→            </Button>

   445→              <Download className="h-4 w-4 mr-2" />
   446→              CSV
   447→            </Button>

   450→              Excel
   451→            </Button>
   452→            <Button variant="outline" size="sm" onClick={() => handleExport('pdf')}>
   453→              <Download className="h-4 w-4 mr-2" />
   454→              PDF
   455→            </Button>
   456→          </div>
   457→        </div>
   458→
   459→        {/* Period Selector */}
   460→        <div className="flex items-center gap-4 flex-wrap">
   461→          <Tabs value={period} onValueChange={(v) => setPeriod(v as any)} className="w-auto">

   477→                {getPeriodLabel()}

   479→              <Button variant="outline" size="icon" onClick={() => navigatePeriod('next')}>
   480→                <ChevronRight className="h-4 w-4" />
   481→              </Button>
   482→            </div>
   483→          )}
   484→        </div>
   485→
   486→        {/* Key Metrics */}
   487→        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-4 gap-4">
   488→          <Card className="border-l-4 border-l-[#00897B]">
   489→            <CardContent className="p-4">
   490→              <div className="flex items-center justify-between">
   491→                <div>
   492→                  <p className="text-xs text-[#757575] mb-1">Total Sales</p>
   493→                  <p className="text-2xl font-bold text-[#1A1A1A]">{formatCurrency(metrics.totalSales)}</p>
   494→                  <div className="flex items-center gap-1 mt-1">
   495→                    {metrics.growth >= 0 ? (
   496→                      <>
   497→                        <TrendingUp className="h-3 w-3 text-green-500" />
   498→                        <span className="text-xs text-green-500">+{metrics.growth.toFixed(1)}%</span>
   499→                      </>
   500→                    ) : (
   501→                      <>
   502→                        <TrendingDown className="h-3 w-3 text-red-500" />
   503→                        <span className="text-xs text-red-500">{metrics.growth.toFixed(1)}%</span>
   504→                      </>
   505→                    )}
   506→                    <span className="text-xs text-[#757575]">vs previous</span>
   507→                  </div>
   508→                </div>
   509→                <div className="w-12 h-12 rounded-full bg-[#E0F2F1] flex items-center justify-center">

   512→              </div>
   513→            </CardContent>
   514→          </Card>
   515→
   516→          <Card>
   517→            <CardContent className="p-4">

   519→                <div>
   520→                  <p className="text-xs text-[#757575] mb-1">Transactions</p>

   522→                  <p className="text-xs text-[#757575] mt-1">
   523→                    {period === 'yearly' ? `${Math.round(metrics.transactionCount / 365)} avg/day` : 
   524→                     period === 'monthly' || period === 'quarterly' ? `${Math.round(metrics.transactionCount / 30)} avg/day` :
   525→                     `${Math.round(metrics.transactionCount / 7)} avg/day`}
   526→                  </p>
   527→                </div>
   528→                <div className="w-12 h-12 rounded-full bg-blue-50 flex items-center justify-center">
   529→                  <ShoppingCart className="h-6 w-6 text-blue-500" />
   530→                </div>
   531→              </div>
   532→            </CardContent>
   533→          </Card>
   534→
   535→          <Card>
   536→            <CardContent className="p-4">
   537→              <div className="flex items-center justify-between">
   538→                <div>
   539→                  <p className="text-xs text-[#757575] mb-1">Avg. Transaction</p>
   540→                  <p className="text-2xl font-bold text-[#1A1A1A]">{formatCurrency(metrics.avgTransaction)}</p>
   541→                  <div className="flex items-center gap-1 mt-1">
   542→                    {comparison.change >= 0 ? (
   543→                      <ArrowUpRight className="h-3 w-3 text-green-500" />
   544→                    ) : (
   545→                      <ArrowDownRight className="h-3 w-3 text-red-500" />
   546→                    )}
   547→                    <span className="text-xs text-[#757575]">vs previous period</span>
   548→                  </div>
   549→                </div>
   550→                <div className="w-12 h-12 rounded-full bg-purple-50 flex items-center justify-center">
   551→                  <CreditCard className="h-6 w-6 text-purple-500" />
   552→                </div>
   553→              </div>
   554→            </CardContent>
   555→          </Card>
   556→
   557→          <Card>
   558→            <CardContent className="p-4">
   559→              <div className="flex items-center justify-between">
   560→                <div>
