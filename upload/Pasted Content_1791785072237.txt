### 13.5 Blockchain (Hyperledger Fabric) Chaincode

```javascript
// SAPE Chaincode — anchors product identity and custody events

'use strict';

const { Contract } = require('fabric-contract-api');

class SAPEContract extends Contract {

  // Anchor product identity
  async anchorProduct(ctx, productIdentityId, gtin, batch, expiry, signature) {
    const identity = {
      productIdentityId,
      gtin,
      batch,
      expiry,
      signature,
      timestamp: new Date().toISOString(),
    };
    await ctx.stub.putState(
      `PRODUCT_${productIdentityId}`,
      Buffer.from(JSON.stringify(identity))
    );
    return JSON.stringify(identity);
  }

  // Anchor custody event
  async anchorCustodyEvent(ctx, eventId, productSerialId, eventType, location, actorId) {
    const event = {
      eventId,
      productSerialId,
      eventType,
      location,
      actorId,
      timestamp: new Date().toISOString(),
    };
    await ctx.stub.putState(
      `CUSTODY_${eventId}`,
      Buffer.from(JSON.stringify(event))
    );
    return JSON.stringify(event);
  }

  // Verify product
  async verifyProduct(ctx, productIdentityId) {
    const identityBytes = await ctx.stub.getState(`PRODUCT_${productIdentityId}`);
    if (!identityBytes || identityBytes.length === 0) {
      throw new Error(`Product ${productIdentityId} not found`);
    }
    return identityBytes.toString();
  }

  // Query custody chain
  async queryCustodyChain(ctx, productSerialId) {
    const query = {
      selector: { productSerialId: productSerialId },
      sort: [{ timestamp: 'asc' }],
    };
    const iterator = await ctx.stub.getQueryResult(JSON.stringify(query));
    const results = [];
    while (true) {
      const res = await iterator.next();
      if (res.value && res.value.value.toString()) {
        results.push(JSON.parse(res.value.value.toString('utf8')));
      }
      if (res.done) {
        await iterator.close();
        return JSON.stringify(results);
      }
    }
  }
}

module.exports = SAPEContract;
```

// Yoco Adapter

class YocoAdapter implements PaymentAdapter {
  private baseUrl = 'https://api.yoco.com/v1';
  private secretKey: string;

  async createPayment(request: CreatePaymentRequest): Promise<CreatePaymentResponse> {
    const response = await fetch(`${this.baseUrl}/checkouts`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.secretKey}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        amount: request.amount,
        currency: 'ZAR',
        successUrl: `${process.env.APP_URL}/payment/success`,
        cancelUrl: `${process.env.APP_URL}/payment/cancel`,
        failureUrl: `${process.env.APP_URL}/payment/failure`,
        metadata: {
          merchantId: request.merchantId,
          reference: request.reference,
        },
      }),
    });

    const data = await response.json();
    return {
      paymentId: data.id,
      provider: 'YOCO',
      providerReference: data.id,
      status: 'PENDING',
      paymentUrl: data.redirectUrl,
      expiresAt: new Date(data.expiresAt),
    };
  }

  async verifyWebhook(payload: any, signature: string): Promise<boolean> {
    const crypto = require('crypto');
    const expectedSignature = crypto
      .createHmac('sha256', process.env.YOCO_WEBHOOK_SECRET)
      .update(JSON.stringify(payload))
      .digest('hex');
    return crypto.timingSafeEqual(
      Buffer.from(signature),
      Buffer.from(expectedSignature)
    );
  }

  async refundPayment(paymentId: string, amount: number): Promise<RefundResponse> {
    const response = await fetch(`${this.baseUrl}/refunds`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.secretKey}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify({
        chargeId: paymentId,
        amount: amount,
      }),
    });
    return response.json();
  }
}

### 16.2 iKhokha

```typescript
// iKhokha Adapter (HMAC-SHA256 signature required)

class iKhokhaAdapter implements PaymentAdapter {
  private baseUrl = 'https://dev.ikhokha.com/api';
  private appId: string;
  private appSecret: string;

  async createPayment(request: CreatePaymentRequest): Promise<CreatePaymentResponse> {
    const timestamp = Date.now().toString();
    const payload = {
      amount: request.amount / 100,
      currency: 'ZAR',
      reference: request.reference,
      merchantId: request.merchantId,
      callbackUrl: `${process.env.APP_URL}/webhooks/ikhokha`,
    };

    const signature = this.generateSignature(payload, timestamp);

    const response = await fetch(`${this.baseUrl}/pay`, {
      method: 'POST',
      headers: {
        'IK-AppID': this.appId,
        'IK-Timestamp': timestamp,
        'IK-Signature': signature,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(payload),
    });

    return response.json();
  }

  private generateSignature(payload: any, timestamp: string): string {
    const crypto = require('crypto');
    const payloadString = JSON.stringify(payload);
    const signatureString = `${this.appId}${timestamp}${payloadString}`;
    return crypto
      .createHmac('sha256', this.appSecret)
      .update(signatureString)
      .digest('hex');
  }
}
```

## §20 FEATURE FLAGS & PROGRESSIVE ROLLOUT

### 20.1 Feature Flag Architecture

```typescript
// Feature Flag Service

interface FeatureFlag {
  key: string;
  enabled: boolean;
  rolloutPercentage: number; // 0-100
  targetIds?: string[]; // Specific doors/merchants
  targetRegions?: string[];
  targetBrands?: string[];
  createdAt: Date;
  updatedAt: Date;
}

class FeatureFlagService {
  async isEnabled(flagKey: string, context: FlagContext): Promise<boolean> {
    const flag = await this.getFlag(flagKey);
    if (!flag || !flag.enabled) return false;
    
    if (flag.targetIds?.includes(context.merchantId)) return true;
    if (flag.targetRegions?.includes(context.region)) return true;
    if (flag.targetBrands?.includes(context.brand)) return true;
    
    // Percentage rollout
    const hash = this.hashContext(flagKey, context.merchantId);
    return (hash % 100) < flag.rolloutPercentage;
  }
}
```
